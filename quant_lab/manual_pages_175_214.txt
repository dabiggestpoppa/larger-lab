from sklearn.cluster import KMeans
import numpy as np, pandas as pd
def discover_tiers(csv_path, start='2024-01-01', end='2026-05-08', pip_size=0.0001):
    df = pd.read_csv(csv_path, header=None,
        names=['dt','open','high','low','close','vol','spread','ts'])
    df['dt'] = pd.to_datetime(df['dt'])
    df = df.set_index('dt').sort_index()
    df = df.tz_localize('UTC').tz_convert('America/New_York')
    df['session'] = (df.index - pd.Timedelta(hours=3)).dt.floor('D')
    ranges = []
    for day, data in df.groupby('session'):
        asian = data[(data.index.hour>=19)|(data.index.hour<3)]
        if len(asian) >= 10:
            ar = (asian['high'].max()-asian['low'].min())/pip_size
            ranges.append(ar)
    ranges = np.array(ranges).reshape(-1,1)
    km = KMeans(n_clusters=3, random_state=42, n_init=10).fit(ranges)
    c = sorted(km.cluster_centers_.flatten())
    return {
        'cutoffs': ((c[0]+c[1])/2, (c[1]+c[2])/2),
        'T1': {'au': int(c[0]*0.5), 'trig': int(c[0]*0.5*1.2)},

        'T3': {'au': int(c[2]*0.5), 'trig': int(c[2]*0.5*1.2)},

import pandas as pd
import numpy as np
from sklearn.cluster import KMeans
def discover_atomic_units(csv_path, pip_size=0.0001):
    df = pd.read_csv(csv_path, header=None,
                     names=['dt','open','high','low','close','vol','spread','ts'])
    df['dt'] = pd.to_datetime(df['dt'])
    df = df.set_index('dt').sort_index()
    if df.index.tz is None:
        df = df.tz_localize('UTC').tz_convert('America/New_York')
    else:
        df = df.tz_convert('America/New_York')
    df['session'] = (df.index - pd.Timedelta(hours=3)).dt.floor('D')
    ranges = []
    for day, data in df.groupby('session'):
        asian = data[(data.index.hour >= 19) | (data.index.hour < 3)]
        if len(asian) >= 10:
            ar = (asian['high'].max() - asian['low'].min()) / pip_size
            ranges.append(ar)
    ranges = np.array(ranges).reshape(-1, 1)
    km = KMeans(n_clusters=3, random_state=42, n_init=10).fit(ranges)
    centroids = sorted(km.cluster_centers_.flatten())
    cutoff1 = (centroids[0] + centroids[1]) / 2
    cutoff2 = (centroids[1] + centroids[2]) / 2
    results = {
        'T1': {'range_max': cutoff1, 'centroid': centroids[0],
               'atomic_unit': int(centroids[0] * 0.50),
               'trigger': int(centroids[0] * 0.50 * 1.2)},
        'T2': {'range_min': cutoff1, 'range_max': cutoff2, 'centroid': centroids[1],
               'atomic_unit': int(centroids[1] * 0.50),
               'trigger': int(centroids[1] * 0.50 * 1.2)},
        'T3': {'range_min': cutoff2, 'centroid': centroids[2],
               'atomic_unit': int(centroids[2] * 0.50),
               'trigger': int(centroids[2] * 0.50 * 1.2)}
    }
    return results
# USAGE: discover_atomic_units('EURUSD_M5.csv', pip_size=0.0001)

def validate_atomic_loop(csv_path, pip_size, tier_configs):
    df = pd.read_csv(csv_path, header=None, names=['dt','open','high','low','close','vol','spread','ts'])
    df['dt'] = pd.to_datetime(df['dt'])
    df = df.set_index('dt').sort_index().tz_localize('UTC').tz_convert('America/New_York')
    df['session'] = (df.index - pd.Timedelta(hours=3)).dt.floor('D')
    hits, misses = 0, 0
    for day, data in df.groupby('session'):
        asian = data[(data.index.hour >= 19) | (data.index.hour < 3)]
        if len(asian) < 10: continue
        ar = (asian['high'].max() - asian['low'].min()) / pip_size
        if ar < tier_configs['T1']['range_max']: tier, au = 'T1', tier_configs['T1']['atomic_unit']
        elif ar < tier_configs['T2']['range_max']: tier, au = 'T2', tier_configs['T2']['atomic_unit']
        else: tier, au = 'T3', tier_configs['T3']['atomic_unit']
        trigger = au * 1.2
        sess = data[(data.index.hour >= 3) & (data.index.hour < 12)]
        if sess.empty: continue
        state, origin = 'SEARCH', None
        for i, r in sess.iterrows():
            if state == 'SEARCH':
                if origin is None: origin = r['close']
                up = (r['high'] - origin) / pip_size
                dn = (origin - r['low']) / pip_size
                if up >= trigger and r['close'] < r['open']: state, origin = 'PB', r['high']
                elif dn >= trigger and r['close'] > r['open']: state, origin = 'PB', r['low']
            elif state == 'PB':
                dz_l = (origin - au * 1.2) if origin == r['high'] else (origin + au * 0.8)
                dz_u = (origin - au * 0.8) if origin == r['high'] else (origin + au * 1.2)
                opp = (r['close'] > r['open']) if origin == r['high'] else (r['close'] < r['open'])
                if dz_l <= r['close'] <= dz_u and opp:
                    entry = r['close']
                    tgt = entry + (au * pip_size) if origin == r['low'] else entry - (au * pip_size)
                    fut = sess[sess.index > i]
                    hit = any((origin == r['low'] and f['high'] >= tgt) or
                              (origin == r['high'] and f['low'] <= tgt) for _, f in fut.iterrows())
                    if hit: hits += 1
                    else: misses += 1
                    state, origin = 'SEARCH', None
    total = hits + misses
    wr = (hits / total) * 100 if total > 0 else 0
    return f"Atomic Loop Validation | Total: {total} | 1 AU Hit Rate: {wr:.1f}%"

                dn = (origin - r['low']) / PIP
                impulse = max(up, dn)

                if impulse >= cfg['trig']:
                    shifted_au = cfg['au']; shift_label = 'NONE'

                    state = 'PB'
                    pb_origin = r['high'] if direction == 1 else r['low']
            elif state == 'PB':
                dz_lo = pb_origin-(shifted_au*1.2*PIP) if direction==1 else pb_origin+(shifted_au*0.8*PIP)
                dz_hi = pb_origin-(shifted_au*0.8*PIP) if direction==1 else pb_origin+(shifted_au*1.2*PIP)
                opp = (r['close']<r['open']) if direction==1 else (r['close']>r['open'])
                in_dz = (dz_lo<=r['close']<=dz_hi) if direction==1 else (dz_hi<=r['close']<=dz_lo)

                    sl = r['high'] if direction==1 else r['low']  # ZERO BUFFER

                            if f['close']<=tgt: tp=True; xi=f.name; break
                            if f['close']>=stop: sl_hit=True; xi=f.name; break
                    trades.append({'tier':tier,'result':'WIN' if tp else 'LOSS' if sl_hit else 'TIME',
                        'pnl_pts':(tgt-entry if tp else stop-entry if sl_hit else f['close']-entry)*dir_})
                    state,origin='SEARCH',entry
    df_t=pd.DataFrame(trades)
    print(f"Trades: {len(df_t)} | WR: {(df_t['result']=='WIN').mean()*100:.1f}% | "
          f"Avg PnL: {df_t['pnl_pts'].mean():+.0f}pts")
    return df_t

# Run on any M5 CSV to discover tiers via k-means

from sklearn.cluster import KMeans
import numpy as np, pandas as pd
def discover_tiers(csv_path, start='2024-01-01', end='2026-05-08', pip_size=0.0001):
    df = pd.read_csv(csv_path, header=None,
        names=['dt','open','high','low','close','vol','spread','ts'])
    df['dt'] = pd.to_datetime(df['dt'])
    df = df.set_index('dt').sort_index()
    df = df.tz_localize('UTC').tz_convert('America/New_York')
    df['session'] = (df.index - pd.Timedelta(hours=3)).dt.floor('D')
    ranges = []
    for day, data in df.groupby('session'):
        asian = data[(data.index.hour>=19)|(data.index.hour<3)]
        if len(asian) >= 10:
            ar = (asian['high'].max()-asian['low'].min())/pip_size
            ranges.append(ar)
    ranges = np.array(ranges).reshape(-1,1)
    km = KMeans(n_clusters=3, random_state=42, n_init=10).fit(ranges)
    c = sorted(km.cluster_centers_.flatten())
    return {
        'cutoffs': ((c[0]+c[1])/2, (c[1]+c[2])/2),
        'T1': {'au': int(c[0]*0.5), 'trig': int(c[0]*0.5*1.2)},
        'T2': {'au': int(c[1]*0.5), 'trig': int(c[1]*0.5*1.2)},
        'T3': {'au': int(c[2]*0.5), 'trig': int(c[2]*0.5*1.2)},
        'note': 'Use recent-only data (2024+) if delta >10% vs full dataset'
    }
