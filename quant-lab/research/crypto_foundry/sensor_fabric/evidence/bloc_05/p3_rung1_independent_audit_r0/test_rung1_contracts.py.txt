"""Independent adversarial probes C01-C22 for the authorized Rung 1 contracts.

MATERIALIZED NAME: ``test_rung1_contracts.py``

Status on the audited tree: IMPLEMENTATION_PENDING.  ``consumer/models.py`` is
not published (the Agent 1 branch is still at the D-1 authorization commit), so
every case below reports NOT_EXECUTED rather than inventing a failure.

Design rules this file holds itself to (the previous suite's conceptual defects
are corrected here):

1. Every case builds a VALID positive control first and asserts it constructs.
   A broken fixture fails loudly as a probe defect, never as a security pass.
2. The attack input actually reaches the entry point the case names.  C05 really
   submits a callable; C04 really submits a verdict object; C03 really injects
   unknown authorization keywords.  No case is satisfied by an ordinary request.
3. An expected validation refusal and an unexpected error are distinguished:
   only ``ValidationError`` / ``ValueError`` / ``TypeError`` count as a refusal;
   anything else is reported as an UNEXPECTED ERROR.
4. Import absence is detected ONCE at module load, and only for the exact
   unpublished module chain.  A missing internal dependency is re-raised, never
   masked.
"""

_MODULE_NAME = "crypto_sensor_fabric.normalization.consumer.models"
_CONSUMER_PACKAGE = "crypto_sensor_fabric.normalization.consumer"


def _load_contracts():
    """Detect implementation absence exactly once, without masking anything else."""
    try:
        return importlib.import_module(_MODULE_NAME), None
    except ModuleNotFoundError as exc:
        missing = exc.name or "<unnamed>"
        # ONLY the unpublished consumer chain may be absent.  Any other missing
        # module is a broken internal dependency and must surface at once.
        if missing not in {_MODULE_NAME, _CONSUMER_PACKAGE}:
            raise
        return None, missing


MODELS, ABSENT_MODULE = _load_contracts()

requires_implementation = pytest.mark.skipif(
    MODELS is None,
    reason=(
        "NOT_EXECUTED — IMPLEMENTATION_PENDING: "
        f"{_MODULE_NAME} is not published on this tree "
        f"(first missing module: {ABSENT_MODULE}); Rung 1 A1.1-A1.5 cannot be "
        "executed and no result is claimed"
    ),
)


def _sym(name: str):
    """A required public symbol.  Absence is an incomplete surface, not a skip."""
    if MODELS is None:  # pragma: no cover - guarded by requires_implementation
        pytest.fail("NOT_EXECUTED — IMPLEMENTATION_PENDING: module absent")
    value = getattr(MODELS, name, None)
    if value is None:
        pytest.fail(
            f"INCOMPLETE SURFACE: {_MODULE_NAME} does not export {name!r}, which "
            "the authorized Rung 1 contract specifies"
        )
    return value


def _request(**payload):
    return _sym("ConsumerRequest")(**payload)


def _valid_request_payload(**overrides) -> dict[str, object]:
    payload: dict[str, object] = {
        "record": fx.valid_trade_record(),
        "artifact_bytes": fx.ARTIFACT_BYTES,
        "required_fields": ("price_canonical",),
        "evidence_references": ((fx.EVIDENCE_REF, fx.EVIDENCE_DIGEST),),
        "knowledge_cutoff": fx.CUTOFF,
        "replay_evidence": fx.valid_replay_evidence(),
        "claimed_manifest_bytes": fx.MANIFEST_BYTES,
    }
    payload.update(overrides)
    return payload


def _accept(build, label: str):
    """A positive control must genuinely construct."""
    try:
        return build()
    except Exception as exc:  # noqa: BLE001 - deliberate: report, never hide
        pytest.fail(
            f"{label}: POSITIVE CONTROL REFUSED — probe/fixture defect, not a "
            f"security finding: {type(exc).__name__}: {exc}"
        )


def _refuse(build, label: str):
    """An attack must be refused by validation, and nothing else."""
    try:
        build()
    except (ValidationError, ValueError, TypeError) as exc:
        return exc
    except Exception as exc:  # noqa: BLE001 - classify, never swallow
        pytest.fail(f"{label}: UNEXPECTED ERROR {type(exc).__name__}: {exc}")
    pytest.fail(f"{label}: attack input was ACCEPTED; no refusal occurred")


def _verdict_kwargs(**overrides):
    payload = {
        "decision": _sym("ConsumerDecision").ACCEPT_FIELD_BINDING,
        "pit_status": "PIT_ADMISSIBLE",
        "replay_status": "REPLAY_ELIGIBLE",
        "custody_status": "CUSTODY_VERIFIED",
        "custody_source_authorization_status": _sym("CustodyAuthorityBasis").INTEGRATION_REQUIRED,
    }
    payload.update(overrides)
    return payload

            **{**base, "verifier": fx.valid_trade_record}

        "ConsumerVerdict",
        "RecordVerification",
        "SemanticAuthority",
        "TrustedAuthorityContext",

        "Callable",
        "Callable[..., ConsumerVerdict]",
    }

    request_cls = _sym("ConsumerRequest")

    for name, field in request_cls.model_fields.items():

            )
        assert name not in forbidden_names, f"ConsumerRequest carries forbidden field {name!r}"

@requires_implementation
def test_c19_model_construct_skips_validation_and_rung1_consumes_nothing():
    request_cls = _sym("ConsumerRequest")

    # The load-bearing Rung 1 law: there is NO consuming operation that could turn
    # an unvalidated instance into authority.
    def _module_level_functions():
        return {
            name: member
            for name, member in inspect.getmembers(MODELS, inspect.isfunction)
            if getattr(member, "__module__", None) == MODELS.__name__

        if any(

        )
    }
    assert consumers == set(), (
        "Rung 1 must not expose a consuming operation; found " f"{sorted(consumers)}"
    )

# C21 -- no hidden I/O and no authority issuance during construction
# ---------------------------------------------------------------------------


@requires_implementation
def test_c21_construction_performs_no_io_and_issues_no_authority(monkeypatch):
    import builtins
    import socket

    events: list[tuple[str, object]] = []
    real_open = builtins.open
    real_time = __import__("time").time

    def spy_open(*args, **kwargs):
        events.append(("open", args[0] if args else None))
        return real_open(*args, **kwargs)

    def spy_socket(*args, **kwargs):
        events.append(("socket", args))
        raise AssertionError("model construction opened a socket")

    monkeypatch.setattr(builtins, "open", spy_open)
    monkeypatch.setattr(socket, "socket", spy_socket)

    before = real_time()
    seam = fx.RootLoaderSeam()
    request = _accept(lambda: _request(**_valid_request_payload()), "C21 control")
    after = real_time()

    assert [event for event in events if event[0] == "open"] == []
    assert seam.load_count == 0, "construction loaded a trust root"
    # No wall-clock read is observable: the fixed cutoff is preserved verbatim.
    assert request.knowledge_cutoff == fx.CUTOFF

    source = Path(MODELS.__file__).read_text(encoding="utf-8")
    for forbidden in ("TRUSTED_MANIFEST_SHA256", "from_reviewed_root", "urandom"):
        assert forbidden not in source, f"contracts reference {forbidden!r}"

    # The authoritative consumer must not be producible by construction: there is
    # no custody/authority field on a request at all.
    assert not [
        name
        for name in type(request).model_fields
        if "authorit" in name or name in {"custody_status", "custody_source_authorized"}
    ]
