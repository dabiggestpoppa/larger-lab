"""Independent fixture architecture for the P3 consumer rungs.

MATERIALIZED NAME: ``rung1_fixtures.py``

Why this file exists
--------------------
An adversarial probe is only as good as its positive control.  A probe whose
fixture is invalid for an unrelated reason "passes" by refusing everything, and
that is indistinguishable from a working defence.  So every fixture here is
built from the REAL ratified record types and is self-tested in ``__main__``.

Two families are deliberately separated (mission section 6):

* ``STRUCTURAL`` fixtures prove that a value of the right shape can be built at
  all.  They are the positive controls for construction tests (C01, C02, C16).
* ``AUTHORIZATION`` fixtures are inputs whose whole purpose is to test a real
  authorization decision (custody bytes, mutable store, root-loader seam).
  They are NOT valid evidence of anything: the trusted custody double holds
  bytes because the verifier hashes them, and the mutable store exists only to
  rebind after the first read.

Nothing here implements a production policy, invents a provider approval, or
touches the shipped manifest.  ``RootLoaderSeam`` is a *recording* double: it
counts every load so a probe can prove a model construction performed no I/O.

Provenance of every constant: derived from the accepted modules
``normalization/models.py``, ``sensors/trade.py``, ``sensors/liquidation.py``,
``time/replay_gate.py`` and ``common/methodologies.py`` at the audited commit.
"""

def digest_of(payload: bytes) -> str:
    """The same digest rule the verifier uses, so a fixture cannot lie by accident."""
    return hashlib.sha256(payload).hexdigest()

# AUTHORIZATION family: doubles, never evidence

class TrustedCustodyDouble:
    """An independently held BYTE store (the ``TrustedCustodyProvider`` shape).

    It holds bytes because the verifier hashes what it is handed.  A double that
    returned a digest would let the verifier believe a report about evidence
    instead of the evidence, which is exactly the P2cR1 F-R1-01 defect.
    """

    def __init__(self, held: dict[str, bytes] | None = None) -> None:
        self._held = dict(held or {EVIDENCE_REF: EVIDENCE_BYTES})
        self.reads: list[str] = []

    def read_evidence(self, reference: str) -> bytes | None:
        self.reads.append(reference)
        return self._held.get(reference)

    def rebind(self, reference: str, payload: bytes) -> None:
        self._held[reference] = payload


class MutableEvidenceStore:
    """A store that changes what it hands over AFTER the first read.

    This is the rebinding attack: a consumer that reads evidence once and then
    reasons about the earlier bytes is defending nothing.
    """

    def __init__(self, first: bytes, second: bytes) -> None:
        self._first = first
        self._second = second
        self._reads = 0

    def read_evidence(self, reference: str) -> bytes | None:
        self._reads += 1
        if self._reads == 1:
            return self._first
        return self._second

    @property
    def reads(self) -> int:
        return self._reads


class RootLoaderSeam:
    """The policy-owned root-loader seam, as a RECORDING double.

    Rung 1 must perform no root load at all.  Rungs 2-3 may load the reviewed
    root exactly through an injected seam like this one, so a probe can count
    loads and prove that construction alone never touched the root.
    """

    def __init__(self, payload: bytes = MANIFEST_BYTES) -> None:
        self._payload = payload
        self.loads: list[str] = []

    def load(self) -> bytes:
        self.loads.append(ROOT_IDENTITY)
        return self._payload

    @property
    def load_count(self) -> int:
        return len(self.loads)

if __name__ == "__main__":  # fixture self-test: every fixture must really build
    import json

    report: dict[str, object] = {}
    for name, factory in (
        ("trade", valid_trade_record),
        ("liquidation", valid_liquidation_record),
        ("replay", valid_replay_evidence),
        ("methodology_registry", methodology_registry),
    ):
        try:
            built = factory()
        except Exception as exc:  # noqa: BLE001 - a broken fixture must be loud
            report[name] = {"state": "INVALID", "error": f"{type(exc).__name__}: {exc}"}
        else:
            report[name] = {"state": "VALID", "type": type(built).__name__}

    report["custody_double"] = {
        "state": "VALID",
        "bytes_held": TrustedCustodyDouble().read_evidence(EVIDENCE_REF) == EVIDENCE_BYTES,
    }
    mutable = MutableEvidenceStore(EVIDENCE_BYTES, EVIDENCE_BYTES_REBOUND)
    report["mutable_store"] = {
        "state": "VALID",
        "rebinds": mutable.read_evidence(EVIDENCE_REF) != mutable.read_evidence(EVIDENCE_REF),
    }
    seam = RootLoaderSeam()
    seam.load()
    report["root_loader_seam"] = {"state": "VALID", "loads_recorded": seam.load_count}

    print(json.dumps(report, indent=2, default=str))
    exit(0 if all(v.get("state") == "VALID" for v in report.values()) else 1)
