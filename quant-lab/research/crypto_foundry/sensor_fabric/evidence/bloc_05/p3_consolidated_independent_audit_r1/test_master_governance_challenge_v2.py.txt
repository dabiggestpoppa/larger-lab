def _git(*args: str) -> str:
    return subprocess.run(
        ["git", "-C", str(AUDIT_QUANT_LAB), *args],
        capture_output=True, encoding="utf-8", errors="replace", check=False,

def _parse_source(text: str) -> ast.Module:
    return ast.parse(text)


def _module_value(tree: ast.Module, name: str) -> Any:
    for node in tree.body:
        if isinstance(node, ast.Assign):
            for target in node.targets:
                if isinstance(target, ast.Name) and target.id == name:
                    return _literal(node.value)
        if isinstance(node, ast.AnnAssign) and isinstance(node.target, ast.Name):
            if node.target.id == name and node.value is not None:
                return _literal(node.value)
    raise LookupError(f"{name!r} is not assigned at module level")


def _literal(node: ast.AST) -> Any:
    """Evaluate a literal allowlist, folding string concatenation only.

    The shipped grants are written as ``frozenset({...})`` and as bare tuples,
    so the container constructor is folded rather than treated as dynamic.
    """
    if isinstance(node, ast.Call) and isinstance(node.func, ast.Name):
        if node.func.id in {"frozenset", "set", "tuple", "list"} and len(node.args) == 1:
            value = _literal(node.args[0])
            if node.func.id == "frozenset":
                return frozenset(value)
            if node.func.id == "set":
                return set(value)
            if node.func.id == "tuple":
                return tuple(value)
            return list(value)
    if isinstance(node, ast.Constant):
        return node.value
    if isinstance(node, ast.BinOp) and isinstance(node.op, ast.Add):
        left, right = _literal(node.left), _literal(node.right)
        if isinstance(left, str) and isinstance(right, str):
            return left + right
        raise ValueError(f"non-string concatenation: {left!r} + {right!r}")
    if isinstance(node, (ast.Tuple, ast.List, ast.Set)):
        values = [_literal(element) for element in node.elts]
        if isinstance(node, ast.Tuple):
            return tuple(values)
        if isinstance(node, ast.Set):
            return set(values)
        return values
    raise ValueError(f"allowlist entry is not a literal: {ast.dump(node)}")


def _legacy_prefixes(audit_path: Path) -> tuple[str, ...]:
    return tuple(_module_value(_parse_source(audit_path.read_text(encoding="utf-8")), "ALLOWED_PREFIXES"))


def _exact_paths(audit_path: Path) -> frozenset[str]:
    tree = _parse_source(audit_path.read_text(encoding="utf-8"))
    try:
        return frozenset(_module_value(tree, "P3_MASTER_EXACT_PATHS"))
    except LookupError:
        return frozenset()


_PYTHON_PATH = os.pathsep.join([str(AUDIT_QUANT_LAB / "src"), str(_NORM)])


def live_admits(audit_module: str, candidates: Sequence[str]) -> dict[str, bool]:
    """Call the SHIPPED predicate.  Behaviour, not a reconstruction of it.

    The audit module is imported in a clean subprocess and its own
    ``_authorized`` is applied to every candidate.  If the predicate is renamed
    or removed this raises rather than silently falling back to an oracle.
    """
    code = (
        "import json, sys\n"
        f"import {audit_module} as law\n"
        "candidates = json.load(sys.stdin)\n"
        "print(json.dumps({c: bool(law._authorized(c)) for c in candidates}))\n"
    )
    result = subprocess.run(
        [sys.executable, "-c", code],
        input=json.dumps(list(candidates)),
        cwd=str(AUDIT_QUANT_LAB),
        capture_output=True, encoding="utf-8", errors="replace", check=False,
        env={**os.environ, "PYTHONPATH": _PYTHON_PATH},
    )
    assert result.returncode == 0, (
        f"the shipped predicate could not be imported from {audit_module}: "
        f"{result.stderr[-1500:]}"
    )
    return json.loads(result.stdout.strip().splitlines()[-1])


def _legacy_prefixes_from_text(text: str) -> tuple[str, ...]:
    """Read the pre-amendment tuple from a historical source text.

    Three shapes exist across the history: module-level ``ALLOWED_PREFIXES``
    (I05 and I10 today), a function-local ``allowed_prefixes`` (I05 at the
    authorization base) and a function-local ``allowed`` (I10 at the base).
    Guessing one of them turned a measurement into a LookupError, which is why
    all three are tried by name.
    """
    tree = _parse_source(text)
    for name in ("ALLOWED_PREFIXES", "allowed_prefixes", "allowed"):
        for node in ast.walk(tree):
            if isinstance(node, ast.Assign):
                for target in node.targets:
                    if isinstance(target, ast.Name) and target.id == name:
                        return tuple(_literal(node.value))
    raise LookupError("no allowlist assignment found in the historical source")


def classify(entry: str) -> str:
    if entry.endswith("/"):
        return "dir"
    if "." in entry.rsplit("/", 1)[-1]:
        return "file"
    return "legacy_prefix"

@pytest.mark.parametrize("which", ["i05", "i10"])
def test_m4_new_exact_paths_are_admitted_and_variants_refuse(which: str) -> None:
    """Requirement: every disclosed path admits, and no variant of it does."""
    audit = I05_AUDIT if which == "i05" else I10_AUDIT
    module = f"test_b5_{which}_scope_audit"
    exact = _exact_paths(audit)
    assert exact == DISCLOSED_EXACT_PATHS, (
        f"{which}: the exact-file grant is not the disclosed set; "
        f"added={sorted(exact - DISCLOSED_EXACT_PATHS)} "
        f"missing={sorted(DISCLOSED_EXACT_PATHS - exact)}"
    )
    prefixes = _legacy_prefixes(audit)
    # A path that already sits inside a historical prefix is admitted by that
    # prefix, not by the exact grant, so its variant is expected to pass.  Those
    # paths are measured separately instead of being quietly excused.
    legacy_covered = sorted(p for p in exact if p.startswith(tuple(prefixes)))
    exact_only = sorted(exact - set(legacy_covered))
    candidates = sorted(exact) + [entry + "_evil.py" for entry in exact_only] + [
        entry + "_evil.py" for entry in legacy_covered
    ]
    verdicts = live_admits(module, candidates)
    refused_variants = [c for c in candidates if c.endswith("_evil.py") and verdicts[c]]
    leaked = [v for v in refused_variants if v not in {
        entry + "_evil.py" for entry in legacy_covered
    }]
    print(f"P3_GRANT {which} " + json.dumps({
        "exact_only": len(exact_only), "legacy_covered": legacy_covered,
        "legacy_covered_variants_admitted": len(refused_variants),
    }, sort_keys=True))
    assert leaked == [], (
        f"{which}: the NEW exact grant still admits a suffix variant: {leaked[:3]}"
    )
    assert legacy_covered == [
        p for p in exact if p.endswith(("test_b5_i08_scope_audit.py", "test_b5_i10_scope_audit.py"))
    ], legacy_covered
    not_admitted = [p for p in exact if not verdicts[p]]
    assert not_admitted == [], f"{which}: granted paths are not admitted: {not_admitted}"


@pytest.mark.parametrize("which", ["i05", "i10"])
def test_m5_exact_admitted_path_set_delta_is_the_grant(which: str) -> None:
    """Requirement: the grant is the delta, and nothing was revoked.

    Measures base predicate vs live predicate over every tracked Python path,
    every disclosed path, every refused variant and the historical samples.
    """
    audit = I05_AUDIT if which == "i05" else I10_AUDIT
    module = f"test_b5_{which}_scope_audit"
    prefixes = _legacy_prefixes(audit)
    universe = sorted(
        {p for p in _git("ls-tree", "-r", "--name-only", "HEAD").splitlines() if p}
        | {p.removeprefix("quant-lab/") for p in
           _git("ls-tree", "-r", "--name-only", AUTHORIZATION_BASE).splitlines() if p}
        | set(DISCLOSED_EXACT_PATHS)
        | set(MUST_REFUSE)
        | set(HISTORICAL_SAMPLES)
    )
    verdicts = live_admits(module, universe)
    base_prefixes = _legacy_prefixes_from_text(
        _git("show", f"{AUTHORIZATION_BASE}:quant-lab/tests/crypto_sensor_fabric/"
                     "normalization/test_b5_i05_scope_audit.py")
    )
    # The base predicate is measured on the I05 tuple only where it applies; for
    # I10 the base tuple is read from I10's own base source.
    if which == "i10":
        base_prefixes = _legacy_prefixes_from_text(
            _git("show", f"{AUTHORIZATION_BASE}:quant-lab/tests/crypto_sensor_fabric/"
                         "normalization/test_b5_i10_scope_audit.py")
        )

    def base(path: str) -> bool:
        if ".." in Path(path).parts:
            return False
        return path.startswith(base_prefixes)

    added = sorted(p for p in universe if verdicts[p] and not base(p))
    removed = sorted(p for p in universe if base(p) and not verdicts[p])
    print(f"PATH_SET_DELTA {which} " + json.dumps({
        "prefix_entries": len(prefixes), "exact_entries": len(_exact_paths(audit)),
        "added": added, "removed": removed, "universe": len(universe),
    }, sort_keys=True))
    assert removed == [], f"{which}: the amendment REVOKED live admission: {removed[:5]}"
    assert set(added) <= DISCLOSED_EXACT_PATHS, (
        f"{which}: the amendment admitted paths beyond the disclosed grant: "
        f"{sorted(set(added) - DISCLOSED_EXACT_PATHS)[:5]}"
    )
    assert added, f"{which}: the grant admitted nothing -- the measurement is vacuous"

def _prefix_admitted_file_entries(prefixes: Iterable[str]) -> list[str]:
    return sorted(e for e in prefixes if classify(e) == "file")


@pytest.mark.parametrize("which", ["i05", "i10"])
def test_m6_historical_prefix_weakness_did_not_grow(which: str) -> None:
    """Requirement: the pre-existing law defect keeps its exact blast radius.

    Every file-kind entry that predates the grant is still tested with
    ``startswith``, so ``<entry>_evil.py`` is admitted.  That weakness is
    preserved by operator directive; what must NOT happen is a NEW entry joining
    it.  The count is compared with the base commit's own tuple.
    """
    audit = I05_AUDIT if which == "i05" else I10_AUDIT
    module = f"test_b5_{which}_scope_audit"
    live_prefixes = _legacy_prefixes(audit)
    base_prefixes = _legacy_prefixes_from_text(
        _git("show", f"{AUTHORIZATION_BASE}:quant-lab/tests/crypto_sensor_fabric/"
                     f"normalization/test_b5_{which}_scope_audit.py")
    )
    live_files = _prefix_admitted_file_entries(live_prefixes)
    base_files = _prefix_admitted_file_entries(base_prefixes)
    variants = [e + "_evil.py" for e in live_files]
    verdicts = live_admits(module, variants) if variants else {}
    offenders = [v for v in variants if verdicts.get(v)]
    print(f"HISTORICAL_PREFIX_WEAKNESS {which} " + json.dumps({
        "live_file_entries": len(live_files), "base_file_entries": len(base_files),
        "prefix_admitted_variants": len(offenders),
        "new_file_entries": sorted(set(live_files) - set(base_files)),
    }, sort_keys=True))
    assert set(live_files) == set(base_files), (
        f"{which}: the grant added prefix-tested file entries "
        f"{sorted(set(live_files) - set(base_files))}"
    )
    assert len(offenders) == len(live_files), (
        f"{which}: the historical weakness is not what was measured: "
        f"{len(offenders)}/{len(live_files)} variants admitted"
    )

def test_m11_non_generated_frozen_artifacts_are_unmodified() -> None:
    """Requirement: hand-written historical evidence is not rewritten."""
    modified: list[str] = []
    for artifact in FROZEN_ARTIFACTS:

        # "unchanged" for every artifact -- a vacuous green.
        rel = f"research/crypto_foundry/sensor_fabric/evidence/bloc_05/{artifact}"

        "republished, so 'unchanged' means the pathspec measured nothing"
    )
    assert offending == [], f"hand-written evidence was rewritten: {offending}"


def test_m12_generated_inventory_republication_is_truthful() -> None:
    """Requirement: a republished generator output still verifies its rows."""
    payload = json.loads(SEM_INVENTORY.read_text(encoding="utf-8"))
    assert payload.get("generator"), "the SEM artifact no longer declares a generator"
    inventory: dict[str, str] = payload["sha256_inventory"]
    stale, missing = [], []
    for rel, digest in inventory.items():
        path = AUDIT_QUANT_LAB / rel
        if not path.is_file():
            missing.append(rel)
            continue
        if hashlib.sha256(path.read_bytes()).hexdigest() != digest:
            stale.append(rel)
    base_payload = json.loads(
        _git("show", f"{AUTHORIZATION_BASE}:quant-lab/research/crypto_foundry/"
                     "sensor_fabric/evidence/bloc_05/bloc_05_semantic_validation.json")
    )
    base_inventory = base_payload["sha256_inventory"]
    moved = sorted(
        rel for rel in inventory
        if base_inventory.get(rel) != inventory[rel]
    )
    added_rows = sorted(set(inventory) - set(base_inventory))
    print("SEM_REPUBLICATION " + json.dumps({
        "rows": len(inventory), "moved": moved, "added_rows": added_rows,
        "stale": stale, "missing": missing,
    }, sort_keys=True))
    assert missing == [] and stale == [], {"stale": stale, "missing": missing}
    assert moved == [
        "tests/crypto_sensor_fabric/normalization/test_b5_i01_scope_audit.py",
        "tests/crypto_sensor_fabric/normalization/test_b5_i05_scope_audit.py",
        "tests/crypto_sensor_fabric/normalization/test_b5_i08_scope_audit.py",
    ], f"undisclosed rows moved: {moved}"

def test_m16_i11r2_progression_is_1111_and_equals_the_measured_count() -> None:
    """Requirement: 1102 -> 1111 is +9 authorized files, not a lost count."""
    payload = json.loads(I11R2_ARTIFACT.read_text(encoding="utf-8"))
    # The I11R2 audit runs `git ls-files '*.py'` from the REPOSITORY ROOT.  From
    # quant-lab it would count only this subtree (696), which is exactly the
    # measurement error the artifact's own history records.
    repo_root = AUDIT_QUANT_LAB.parent
    tracked = [
        l for l in subprocess.run(
            ["git", "-C", str(repo_root), "ls-files", "*.py"],
            capture_output=True, encoding="utf-8", errors="replace", check=False,
        ).stdout.splitlines() if l.strip()
    ]
    base_tracked = [
        l for l in subprocess.run(
            ["git", "-C", str(repo_root), "ls-tree", "-r", "--name-only", AUTHORIZATION_BASE],
            capture_output=True, encoding="utf-8", errors="replace", check=False,
        ).stdout.splitlines() if l.endswith(".py")
    ]
    added = sorted(
        p.removeprefix("quant-lab/") for p in set(tracked) - set(base_tracked)
    )
    expected_added = sorted(
        p for p in DISCLOSED_EXACT_PATHS if p.endswith(".py")
        and not p.startswith("tests/crypto_sensor_fabric/normalization/test_b5_i10")
        and not p.startswith("tests/crypto_sensor_fabric/normalization/test_b5_i08")
    )
    print("I11R2 " + json.dumps({
        "artifact": payload.get("python_files_scanned"),
        "measured": len(tracked), "base": len(base_tracked),
        "added_count": len(added), "added": added,
    }, sort_keys=True))
    assert payload["python_files_scanned"] == len(tracked), (
        payload["python_files_scanned"], len(tracked)
    )
    assert len(tracked) == len(base_tracked) + 9, (len(base_tracked), len(tracked))
    assert added == expected_added, {"added": added, "expected": expected_added}
    assert payload.get("unexpected_hits") == {}, payload.get("unexpected_hits")

def _audit_run(paths: Sequence[Path]) -> subprocess.CompletedProcess:
    return subprocess.run(
        [sys.executable, "-m", "pytest", *[str(p) for p in paths],
         "-q", "-p", "no:cacheprovider"],
        cwd=str(AUDIT_QUANT_LAB), capture_output=True, encoding="utf-8",
        errors="replace", check=False,
        env={**os.environ, "PYTHONPATH": str(AUDIT_QUANT_LAB / "src")},
    )

    """Behavioural: the amended seal accepts the tree it is asked about."""
    result = _audit_run([I08_AUDIT])

    tail = result.stdout.strip().splitlines()[-1] if result.stdout.strip() else ""
    assert result.returncode == 0, f"I08 refused the R5 tree: {tail}\n{result.stderr[-1200:]}"

def test_m21_lineage_is_present_ancestral_and_scope_clean() -> None:
    shas = [sha for _, sha in LINEAGE]
    for sha in shas:
        assert subprocess.run(
            ["git", "-C", str(AUDIT_QUANT_LAB), "cat-file", "-e", f"{sha}^{{commit}}"],
            capture_output=True, check=False,
        ).returncode == 0, f"commit is not present locally: {sha}"
    for earlier, later in zip(shas, shas[1:]):
        ancestor = subprocess.run(
            ["git", "-C", str(AUDIT_QUANT_LAB), "merge-base", "--is-ancestor", earlier, later],
            capture_output=True, check=False,
        )
        assert ancestor.returncode == 0, f"{earlier[:9]} is not an ancestor of {later[:9]}"
    head = _git("rev-parse", "HEAD").strip()
    assert head == LINEAGE[-1][1], f"audited tree is not R5: {head}"

    i05 = _legacy_prefixes(I05_AUDIT)
    per_rung: dict[str, list[str]] = {}
    for (name, sha), (_, previous) in zip(LINEAGE[1:], LINEAGE):
        changed = [
            l.removeprefix("quant-lab/")
            for l in _git("diff", "--name-only", previous, sha).splitlines() if l.strip()
        ]
        per_rung[name] = changed
        code_paths = [p for p in changed if p.startswith(("src/", "tests/"))]
        unauthorized = [p for p in code_paths if not (p.startswith(i05) or p in DISCLOSED_EXACT_PATHS)]
        assert unauthorized == [], f"{name} touched unauthorized code paths: {unauthorized}"
    print("LINEAGE " + json.dumps(per_rung, sort_keys=True))
