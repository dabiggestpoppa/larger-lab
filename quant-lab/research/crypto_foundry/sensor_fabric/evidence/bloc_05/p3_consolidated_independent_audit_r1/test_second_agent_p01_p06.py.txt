class Custody:
    """Instrumented custody double.  Records every read, by reference."""

    def __init__(self, held: dict[str, bytes] | None = None) -> None:
        self._held = {REF: DOC, REF2: DOC} if held is None else dict(held)
        self.reads: list[str] = []

    def read_evidence(self, reference: str) -> bytes | None:
        self.reads.append(reference)
        return self._held.get(reference)


class FlipCustody:
    """Bytes are good on the FIRST read of a reference and `later` afterwards.

    The counter is per reference, not global: the verifier reads every approved
    reference once, then the consumer re-reads the caller's references.  A
    global counter would make the second *approved* blob look tampered during
    verification and measure CUSTODY_UNVERIFIED instead of the rebinding law.
    """

    def __init__(self, later: bytes | None) -> None:
        self.later = later
        self.reads: list[str] = []
        self._counts: dict[str, int] = {}

    def read_evidence(self, reference: str) -> bytes | None:
        self.reads.append(reference)
        self._counts[reference] = self._counts.get(reference, 0) + 1
        return DOC if self._counts[reference] == 1 else self.later


class RaisingCustody:
    def __init__(self, message: str = SENTINEL) -> None:
        self.message = message
        self.reads = 0

    def read_evidence(self, reference: str) -> bytes | None:
        self.reads += 1
        raise RuntimeError(self.message)


class LateRaisingCustody:
    """Succeeds during verification, raises on the post-verification re-read."""

    def __init__(self) -> None:
        self.reads = 0
        self._counts: dict[str, int] = {}

    def read_evidence(self, reference: str) -> bytes | None:
        self.reads += 1
        self._counts[reference] = self._counts.get(reference, 0) + 1
        if self._counts[reference] == 1:
            return DOC
        raise RuntimeError(SENTINEL)


class EmptyCustody:
    def read_evidence(self, reference: str) -> bytes | None:
        return None


class Source:
    def __init__(self, basis: CustodyAuthorityBasis, detail: str = "fixture basis") -> None:
        self.basis = basis
        self.detail = detail
        self.seen: list[CustodyAuthorityIdentity] = []

    def authorize(self, identity: CustodyAuthorityIdentity) -> CustodyAuthorityDerivation:
        self.seen.append(identity)
        return CustodyAuthorityDerivation(basis=self.basis, detail=self.detail)

def verdict_text(verdict: ConsumerVerdict) -> str:
    return json.dumps(
        {name: (None if getattr(verdict, name) is None else str(getattr(verdict, name)))
         for name in verdict.model_fields}
    ) + repr(verdict)


def refused(verdict: ConsumerVerdict, reason: ConsumerRefusalReason | None = None) -> bool:
    if verdict.decision is not ConsumerDecision.REFUSE:
        return False
    return reason is None or verdict.refusal_reason is reason

def test_p01a_a_custody_exception_message_does_not_reach_the_verdict(
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    """Requirement: a bounded diagnostic carries no payload from a dependency.

    Attack: the trusted custody provider raises with a sentinel in its message.
    Control: the same request with a well-behaved provider reaches the pipeline.
    """
    manifest = wire_manifest()
    record = trade_record(monkeypatch)
    control = policy(manifest).evaluate(request(record))
    assert control.decision is ConsumerDecision.INTEGRATION_REQUIRED

    verdict = policy(manifest, custody=RaisingCustody()).evaluate(request(record))
    assert verdict.decision is ConsumerDecision.REFUSE
    assert verdict.refusal_reason is ConsumerRefusalReason.VERIFIER_CRASH
    leaked = SENTINEL in verdict_text(verdict)
    print("P01A " + json.dumps({
        "reason": verdict.refusal_reason.value,
        "detail": (verdict.refusal_detail or "")[:120],
        "sentinel_leaked": leaked,
    }))
    assert not leaked, (
        "LEAK: the custody exception message is relayed verbatim into the public "
        f"refusal_detail: {verdict.refusal_detail!r}"
    )


def test_p01b_a_late_custody_failure_is_bounded_and_silent(

def test_p01c_artifact_and_evidence_bytes_are_not_echoed(
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    """A sentinel inside the caller's own bytes must not come back in a verdict."""

def test_p02b_replay_evidence_of_another_record_is_not_this_records_admissibility(
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    """Requirement: replay admissibility is bound to the record under evaluation.

    Attack: otherwise-eligible evidence whose selected revision belongs to a
    DIFFERENT record and whose market availability is after the cutoff.
    """
    record = trade_record(monkeypatch, revision="rev-OWN")
    foreign = replay(
        market_available_at=datetime(2027, 1, 1, tzinfo=UTC),
        revision=RevisionSelection(
            mode=RevisionMode.AS_KNOWN_THEN,
            selected=(
                RevisionRecord(
                    source_revision_id="rev-FOREIGN",
                    normalization_generation=9,
                    supersedes_t1_record_id=None,
                    revision_status=RevisionStatus.ORIGINAL,
                    known_from=datetime(2027, 1, 1, tzinfo=UTC),
                    known_to=None,
                ),
            ),
            excluded=(), pit_safe=True, pit_uncertain=False,
        ),
    )
    verdict = policy(
        wire_manifest(), source=Source(AUTHORIZED_SOURCE)
    ).evaluate(request(record, replay_evidence=foreign))

    evidence_revision = foreign.revision.selected[0].source_revision_id
    observed = {
        "decision": verdict.decision.value,
        "replay_status": verdict.replay_status,
        "record_revision": record.source_revision_id,
        "evidence_revision": evidence_revision,
        "evidence_available_at": foreign.market_available_at.isoformat(),
        "cutoff": CUTOFF.isoformat(),
    }
    print("P02B " + json.dumps(observed, sort_keys=True))
    assert evidence_revision == record.source_revision_id, (
        "REPLAY_EVIDENCE_NOT_RECORD_BOUND: the policy adjudicated evidence for "
        f"{evidence_revision!r} while evaluating record {record.source_revision_id!r}, "
        f"and reported replay_status={verdict.replay_status!r} "
        f"with decision={verdict.decision.value}"
    )
    assert foreign.market_available_at <= CUTOFF, (
        "REPLAY_EVIDENCE_AVAILABLE_AFTER_CUTOFF: the policy admitted evidence "
        "that only became available after the request's own knowledge cutoff"
    )

def test_p03c_the_derivation_binds_every_identity_element(
    monkeypatch: pytest.MonkeyPatch, field: str, value: str
) -> None:
    """Requirement: authority does not travel across an identity axis.

    The root must be pinned to this fixture first, otherwise EVERY identity is
    refused at the first check and the case would pass without ever reaching the
    identity laws (the same vacuity a missing positive control always hides).
    """
    manifest = wire_manifest()
    pin(monkeypatch, manifest)
    digest = hashlib.sha256(manifest).hexdigest()
    base = dict(
        manifest_sha256=digest, artifact_id=ARTIFACT_ID, approval_id=APPROVAL_ID,
        registry_digest=ARTIFACT_SHA, record_kind="TRADE",
        provider="BINANCE_USDM", knowledge_cutoff=CUTOFF,
    )
    # positive control: the unmodified identity passes every binding this schema
    # can check and stops at the unresolved authority contract, not at a refusal.
    control = derive_custody_authority(
        manifest_bytes=manifest, identity=CustodyAuthorityIdentity(**base)
    )
    assert control.basis is CustodyAuthorityBasis.INTEGRATION_REQUIRED, control.detail
    if field == "record_kind":
        # A wildcard kind is refused by the type law, not by the derivation.
        with pytest.raises(Exception):
            CustodyAuthorityIdentity(**{**base, field: value})
        return
    identity = CustodyAuthorityIdentity(**{**base, field: value})
    derivation = derive_custody_authority(manifest_bytes=manifest, identity=identity)
    assert derivation.basis is CustodyAuthorityBasis.UNAUTHORIZED, (
        field, derivation.basis, derivation.detail
    )

def test_p04a_every_stage_returns_a_verdict_for_hostile_data(
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    """Requirement: ``evaluate`` never raises for data, at any stage.

    Each case bypasses validation with ``model_copy``/``model_construct`` so the
    policy's own revalidation and later stages are the only defence.
    """
    record = trade_record(monkeypatch)
    manifest = wire_manifest()
    valid = request(record)
    cases = {
        "naive_cutoff": valid.model_copy(update={"knowledge_cutoff": datetime(2026, 6, 1, 12)}),
        "bare_date_cutoff": valid.model_copy(update={"knowledge_cutoff": "2026-06-01"}),
        "record_is_mapping": valid.model_copy(update={"record": {"a": 1}}),
        "record_is_object": valid.model_copy(update={"record": object()}),
        "record_is_none": valid.model_copy(update={"record": None}),
        "artifact_is_str": valid.model_copy(update={"artifact_bytes": "not bytes"}),
        "evidence_is_object": valid.model_copy(update={"evidence_references": (object(),)}),
        "replay_is_string": valid.model_copy(update={"replay_evidence": "eligible"}),
        "claimed_is_str": valid.model_copy(update={"claimed_manifest_bytes": "x"}),
    }
    escaped: dict[str, str] = {}
    decisions: dict[str, str] = {}
    for name, hostile in cases.items():
        try:
            verdict = policy(manifest).evaluate(hostile)
        except Exception as exc:  # noqa: BLE001 - the measurement is the escape
            escaped[name] = f"{type(exc).__name__}: {exc}"
            continue
        decisions[name] = verdict.decision.value
        assert verdict.decision is not ConsumerDecision.ACCEPT_FIELD_BINDING, name
    print("P04A " + json.dumps({"escaped": escaped, "decisions": decisions}, sort_keys=True))
    assert escaped == {}, f"UNEXPECTED_EXCEPTION outside the fail-closed boundary: {escaped}"
