def _rev(revision_id: str = REV, known_from: object = EARLY) -> RevisionRecord:
    return RevisionRecord(
        source_revision_id=revision_id,
        normalization_generation=1,
        supersedes_t1_record_id=None,
        revision_status=RevisionStatus.ORIGINAL,
        known_from=known_from,
        known_to=None,
    )


def _broken(revision_id: str = "rev-BROKEN") -> RevisionRecord:
    """A revision whose knowledge instant is not an instant (hostile data)."""
    return RevisionRecord.model_construct(
        source_revision_id=revision_id,
        normalization_generation=1,
        supersedes_t1_record_id=None,
        revision_status=RevisionStatus.ORIGINAL,
        known_from="not-an-instant",
        known_to=None,
    )


def _selection(
    *revs: RevisionRecord, excluded: tuple[RevisionRecord, ...] = ()
) -> RevisionSelection:
    return RevisionSelection(
        mode=RevisionMode.AS_KNOWN_THEN,
        selected=tuple(revs) if revs else (_rev(),),
        excluded=excluded,
        pit_safe=True,
        pit_uncertain=False,
    )


class _Custody:
    def __init__(self, payloads: object = None) -> None:
        self._payloads = R5.DOC if payloads is None else payloads
        self.reads: list[str] = []

    def read_evidence(self, reference: str) -> object:
        self.reads.append(reference)
        return self._payloads


class _AuthSource:
    """The test-only composition control: a source that WOULD authorize."""

    def __init__(self, echos: BaseException | None = None) -> None:
        self._echos = echos
        self.seen: list[object] = []

    def authorize(self, identity: object) -> CustodyAuthorityDerivation:
        if self._echos is not None:
            raise self._echos
        self.seen.append(identity)
        return CustodyAuthorityDerivation(
            basis=CustodyAuthorityBasis.AUTHORIZED,
            detail="probe authority",
            identity_sha256=custody_authority_identity_digest(identity),
        )


def _text(verdict: object) -> str:
    return R5.verdict_text(verdict)


def _no_sentinel(verdict: object) -> None:
    assert SENTINEL not in _text(verdict), _text(verdict)


def _policy(manifest: bytes | None = None, **kwargs: object):
    return R5.policy(R5.wire_manifest() if manifest is None else manifest, **kwargs)


def _evidence(selection: RevisionSelection):
    return R5.replay(revision=selection)


def _evaluate(monkeypatch, selection: RevisionSelection, **kwargs: object):
    """Reach the accept path where the selection alone decides the outcome."""
    record = R5.trade_record(monkeypatch)
    evidence = _evidence(selection)
    kwargs.setdefault("source", _AuthSource())
    return _policy(**kwargs).evaluate(
        R5.request(record, replay_evidence=evidence, knowledge_cutoff=CUTOFF)
    )


def _refused_temporally(verdict: object) -> None:
    """A refusal that must be the temporal control, not a downstream stage."""
    text = _text(verdict)
    assert verdict.decision is REFUSE, text
    assert verdict.decision is not ConsumerDecision.INTEGRATION_REQUIRED, text
    assert verdict.refusal_reason is ConsumerRefusalReason.PIT_INADMISSIBLE, text
    assert verdict.replay_status == "NOT_BOUND", text
    assert "LATER cutoff" in (verdict.refusal_detail or ""), text

@pytest.mark.parametrize(
    "order",
    ["future_last", "future_first"],
    ids=["future_last", "future_first"],
)
def test_f4_08_contradictory_duplicate_never_certifies(monkeypatch, order):
    """Two knowledge instants for one id: no ordering of them may certify."""
    pair = (
        (_rev(REV, EARLY), _rev(REV, FUTURE))
        if order == "future_last"
        else (_rev(REV, FUTURE), _rev(REV, EARLY))

        pytest.fail(
            f"a contradictory duplicate selection ({order}) was accepted: "
            + _text(verdict)
        )
    assert verdict.refusal_reason in {
        ConsumerRefusalReason.PIT_INADMISSIBLE,
        ConsumerRefusalReason.REPLAY_BLOCKED,
    }, _text(verdict)

def test_f4_10_malformed_knowledge_instant_on_another_entry(monkeypatch):
    """Either the temporal control refuses it, or the boundary guard does.

    The important part is which path ran: a schema-guard refusal is NOT credit
    for the temporal control, and the assertion says so.
    """
    verdict = _evaluate(monkeypatch, _selection(_rev(), _broken()))
    detail = verdict.refusal_detail or ""
    assert verdict.decision is REFUSE, _text(verdict)

        assert "LATER cutoff" in detail, _text(verdict)
    else:
        assert verdict.refusal_reason is ConsumerRefusalReason.VERIFICATION_FAILED, _text(
            verdict
        )
        assert "does not satisfy its own schema" in detail, _text(verdict)


def test_f4_11_order_alone_may_not_decide_the_outcome(monkeypatch):

    )

def test_b_06_the_named_limitation_is_a_constant_not_an_enforcement_branch(
    monkeypatch,
):
    """Operation, not documentation: removing the constant changes nothing."""
    source = (
        QUANT_LAB / "src/crypto_sensor_fabric/normalization/consumer/policy.py"
    ).read_text(encoding="utf-8")
    tree = ast.parse(source)
    referenced = [
        node.lineno
        for node in ast.walk(tree)
        if (
            isinstance(node, ast.Name)
            and node.id == "UNRESOLVED_REPLAY_BINDING"
            or isinstance(node, ast.Attribute)
            and node.attr == "UNRESOLVED_REPLAY_BINDING"
        )
    ]
    assert referenced, "the named limitation disappeared from the module"
    for node in ast.walk(tree):
        if isinstance(node, ast.If):
            names = {
                child.id
                for child in ast.walk(node.test)
                if isinstance(child, ast.Name)
            }
            assert "UNRESOLVED_REPLAY_BINDING" not in names, (
                f"line {node.lineno}: the constant became a runtime branch"
            )

    record = R5.trade_record(monkeypatch)
    evidence = R5.replay(source_evidence_refs=("ANOTHER_PROVIDER_TIMESTAMP_DOC",))

    def outcome() -> str:
        verdict = _policy(source=_AuthSource()).evaluate(
            R5.request(record, replay_evidence=evidence)
        )
        return verdict.decision.value

    before = outcome()
    monkeypatch.setattr(policy_module, "UNRESOLVED_REPLAY_BINDING", ())
    after = outcome()
    assert before == after == ACCEPT.value, (
        f"the constant has runtime effect ({before} -> {after}); re-classify it as a "
        "control and say what it enforces"
    )
