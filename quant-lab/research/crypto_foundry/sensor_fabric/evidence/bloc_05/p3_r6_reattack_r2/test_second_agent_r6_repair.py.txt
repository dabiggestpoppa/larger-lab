def _rev(
    revision_id: str = REV,
    known_from: object = EARLY,
    generation: int = 1,
) -> RevisionRecord:
    return RevisionRecord(
        source_revision_id=revision_id,
        normalization_generation=generation,
        supersedes_t1_record_id=None,
        revision_status=RevisionStatus.ORIGINAL,
        known_from=known_from,
        known_to=None,
    )


def _selection(
    *revs: RevisionRecord, excluded: tuple[RevisionRecord, ...] = ()
) -> RevisionSelection:
    return RevisionSelection(
        mode=RevisionMode.AS_KNOWN_THEN,
        selected=tuple(revs) if revs else (_rev(),),
        excluded=excluded,
        pit_safe=True,
        pit_uncertain=False,
    )


class _Custody:
    def __init__(self, payloads: object = None, exc: BaseException | None = None) -> None:
        self._payloads = R5.DOC if payloads is None else payloads
        self._exc = exc
        self.reads: list[str] = []

    def read_evidence(self, reference: str) -> object:
        self.reads.append(reference)
        if self._exc is not None:
            raise self._exc
        return self._payloads


class _AuthSource:
    """A custody source that WOULD authorize when the root approves artifacts."""

    def __init__(self, echos: BaseException | None = None) -> None:
        self._echos = echos
        self.seen: list[object] = []

    def authorize(self, identity: object) -> CustodyAuthorityDerivation:
        if self._echos is not None:
            raise self._echos
        self.seen.append(identity)
        return CustodyAuthorityDerivation(
            basis=CustodyAuthorityBasis.AUTHORIZED,
            detail="probe authority",
            identity_sha256=custody_authority_identity_digest(identity),
        )


class _ForgedSource:
    def __init__(self, basis: CustodyAuthorityBasis, digest: str | None) -> None:
        self._basis = basis
        self._digest = digest

    def authorize(self, identity: object) -> CustodyAuthorityDerivation:
        return CustodyAuthorityDerivation.model_construct(
            basis=self._basis, detail="forged", identity_sha256=self._digest
        )


class _WrongIdentity:
    def authorize(self, identity: object) -> CustodyAuthorityDerivation:
        return CustodyAuthorityDerivation(
            basis=CustodyAuthorityBasis.AUTHORIZED,
            detail="another identity",
            identity_sha256="0" * 64,
        )


def _text(verdict: object) -> str:

def _no_sentinel(verdict: object) -> None:

    ).read_text(encoding="utf-8")

    for node in ast.walk(tree):
        if isinstance(node, ast.Name) and node.id == "UNRESOLVED_REPLAY_BINDING":
            uses.append(node.lineno)
        if isinstance(node, ast.Attribute) and node.attr == "UNRESOLVED_REPLAY_BINDING":
            uses.append(node.lineno)
    assert uses, "the named limitation disappeared from the module"
    for node in ast.walk(tree):

            names = {
                child.id
                for child in ast.walk(node)
                if isinstance(child, ast.Name)

            assert "UNRESOLVED_REPLAY_BINDING" not in names, (
                f"line {node.lineno}: the named limitation became an enforcement "
                "branch without a schema change"
