def _rev(revision_id: str = REV, known_from: object = EARLY) -> RevisionRecord:
    return RevisionRecord(
        source_revision_id=revision_id,
        normalization_generation=1,
        supersedes_t1_record_id=None,
        revision_status=RevisionStatus.ORIGINAL,
        known_from=known_from,
        known_to=None,
    )


def _broken(revision_id: str = "rev-BROKEN") -> RevisionRecord:
    """A revision whose knowledge instant is not an instant (hostile data)."""
    return RevisionRecord.model_construct(
        source_revision_id=revision_id,
        normalization_generation=1,
        supersedes_t1_record_id=None,
        revision_status=RevisionStatus.ORIGINAL,
        known_from="not-an-instant",
        known_to=None,
    )


def _selection(
    *revs: RevisionRecord, excluded: tuple[RevisionRecord, ...] = ()
) -> RevisionSelection:
    return RevisionSelection(
        mode=RevisionMode.AS_KNOWN_THEN,
        selected=tuple(revs) if revs else (_rev(),),
        excluded=excluded,
        pit_safe=True,
        pit_uncertain=False,
    )


class _Custody:
    def __init__(self, payloads: object = None) -> None:
        self._payloads = R5.DOC if payloads is None else payloads
        self.reads: list[str] = []

    def read_evidence(self, reference: str) -> object:
        self.reads.append(reference)
        return self._payloads


class _AuthSource:
    """The test-only composition control: a source that WOULD authorize."""

    def __init__(self, echos: BaseException | None = None) -> None:
        self._echos = echos
        self.seen: list[object] = []

    def authorize(self, identity: object) -> CustodyAuthorityDerivation:
        if self._echos is not None:
            raise self._echos
        self.seen.append(identity)
        return CustodyAuthorityDerivation(
            basis=CustodyAuthorityBasis.AUTHORIZED,
            detail="probe authority",
            identity_sha256=custody_authority_identity_digest(identity),
        )


def _text(verdict: object) -> str:
    return R5.verdict_text(verdict)


def _no_sentinel(verdict: object) -> None:
    assert SENTINEL not in _text(verdict), _text(verdict)


def _policy(manifest: bytes | None = None, **kwargs: object):
    return R5.policy(R5.wire_manifest() if manifest is None else manifest, **kwargs)


def _evidence(selection: RevisionSelection):
    return R5.replay(revision=selection)


def _evaluate(monkeypatch, selection: RevisionSelection, **kwargs: object):
    """PRODUCTION DEFAULT: the module guard is untouched."""
    assert getattr(policy_module, BINDING_GUARD) is False, (
        "the production default must leave the binding guard off; this probe "
        "measures the production default and a flipped guard would invalidate it"
    )
    record = R5.trade_record(monkeypatch)
    evidence = _evidence(selection)
    kwargs.setdefault("source", _AuthSource())
    return _policy(**kwargs).evaluate(
        R5.request(record, replay_evidence=evidence, knowledge_cutoff=CUTOFF)
    )


def _evaluate_with_override(monkeypatch, selection: RevisionSelection, **kwargs: object):
    """LABELLED TEST-ONLY OVERRIDE -- exercises the accepting machinery only."""
    monkeypatch.setattr(policy_module, BINDING_GUARD, True)
    record = R5.trade_record(monkeypatch)
    evidence = _evidence(selection)
    kwargs.setdefault("source", _AuthSource())
    return _policy(**kwargs).evaluate(
        R5.request(record, replay_evidence=evidence, knowledge_cutoff=CUTOFF)
    )


def _assert_intermediate_temporal_pass(verdict: object) -> None:
    """The temporal control ADMITTED this selection; the guard is what refused.

    This is the §1 anchor.  ``INTEGRATION_REQUIRED`` alone would be ambiguous --
    it is also what a broken harness produces -- so the assertion additionally
    requires the replay stage to report ELIGIBLE and requires the terminal detail
    to name the unmet binding contract rather than any temporal rule.
    """
    text = _text(verdict)
    assert verdict.decision is INTEGRATION, text
    assert verdict.replay_status == "ELIGIBLE", text
    detail = verdict.refusal_detail or ""
    assert BINDING_CONTRACT in text, text
    assert "LATER cutoff" not in detail, text
    assert verdict.refusal_reason is None, text


def _assert_test_only_override_reaches_acceptance(verdict: object) -> None:
    text = _text(verdict)
    assert verdict.decision is ACCEPT, (
        "the accepting machinery is unreachable through the labelled test-only "
        "override, so a refusals-only matrix cannot be distinguished from a "
        "broken harness: " + text
    )
    assert verdict.replay_status == "ELIGIBLE", text


def _refused_temporally(verdict: object) -> None:
    """A refusal that must be the temporal control, not a downstream stage."""
    text = _text(verdict)
    assert verdict.decision is REFUSE, text
    assert verdict.decision is not INTEGRATION, text
    assert verdict.refusal_reason is ConsumerRefusalReason.PIT_INADMISSIBLE, text
    assert verdict.replay_status == "NOT_BOUND", text
    assert "LATER cutoff" in (verdict.refusal_detail or ""), text


def _refused_as_contradiction(verdict: object) -> None:
    """A duplicate or self-contradictory selection refuses as a CONTRADICTION.

    §2 requires that a duplicate identifier cannot conceal future information.
    The repair refuses the selection before ordering it, which is the stronger
    answer; what matters for attribution is that the refusal is the rebinding
    stage's own (REPLAY_BLOCKED) and not the terminal integration guard nor a
    verification-schema accident.
    """
    text = _text(verdict)
    assert verdict.decision is REFUSE, text
    assert verdict.decision is not INTEGRATION, text
    assert verdict.refusal_reason is ConsumerRefusalReason.REPLAY_BLOCKED, text

def test_f4_10_malformed_knowledge_instant_on_another_entry(monkeypatch):
    """Either the temporal control refuses it, or the boundary guard does.

    The important part is which path ran: a schema-guard refusal is NOT credit
    for the temporal control, and the assertion says so.
    """
    verdict = _evaluate(monkeypatch, _selection(_rev(), _broken()))
    detail = verdict.refusal_detail or ""
    assert verdict.decision is REFUSE, _text(verdict)
    if verdict.refusal_reason is ConsumerRefusalReason.PIT_INADMISSIBLE:

        assert "not a timestamp" in detail, _text(verdict)
    else:
        assert verdict.refusal_reason is ConsumerRefusalReason.VERIFICATION_FAILED, _text(
            verdict
        )
        assert "does not satisfy its own schema" in detail, _text(verdict)


def test_f4_11_order_alone_may_not_decide_the_outcome(monkeypatch):

    )
    _refused_as_contradiction(forward)
    _refused_as_contradiction(reverse)
    assert forward.refusal_reason is reverse.refusal_reason

def _unapproved_refusal() -> SemanticAuthorityRefusal:
    """A SemanticAuthorityRefusal whose code is NOT in the approved vocabulary.

    Built without running ``__init__`` because that constructor formats
    ``refusal.value``; the point here is a refusal-shaped object whose ``value``
    is arbitrary text.  If the policy echoed it, the leak would be the payload.
    """

    class _NotApproved:
        value = SENTINEL

    exc = SemanticAuthorityRefusal.__new__(SemanticAuthorityRefusal)
    exc.refusal = _NotApproved()  # type: ignore[assignment]
    exc.detail = SENTINEL
    ValueError.__init__(exc, SENTINEL)
    return exc

def test_b_10_the_binding_guard_is_executable_and_the_old_constant_is_not(
    monkeypatch,
):
    """Operation, not documentation.

    The historical finding stands for ``UNRESOLVED_REPLAY_BINDING``: it is a
    string and no branch reads it.  The ENFORCED control is the new boolean, and
    it must be demonstrably executable -- flipping it must change the outcome.
    """
    source = (
        QUANT_LAB / "src/crypto_sensor_fabric/normalization/consumer/policy.py"
    ).read_text(encoding="utf-8")
    tree = ast.parse(source)

    documented = [
        node.lineno
        for node in ast.walk(tree)
        if isinstance(node, ast.Name) and node.id == "UNRESOLVED_REPLAY_BINDING"
    ]
    assert documented, "the named limitation disappeared from the module"
    for node in ast.walk(tree):
        if isinstance(node, ast.If):
            names = {
                child.id
                for child in ast.walk(node.test)
                if isinstance(child, ast.Name)
            }
            assert "UNRESOLVED_REPLAY_BINDING" not in names, (
                f"line {node.lineno}: the documentation constant became a branch"
            )

    enforced = [
        node.lineno
        for node in ast.walk(tree)
        if isinstance(node, ast.If)
        and any(
            isinstance(child, ast.Name) and child.id == BINDING_GUARD
            for child in ast.walk(node.test)
        )
    ]
    assert enforced, "the executable binding guard is not read by any branch"
    assert getattr(policy_module, BINDING_GUARD) is False, (
        "production must ship the guard off; a flipped default would be an "
        "acceptance the schema cannot support"
    )
    assert getattr(policy_module, "REPLAY_EVIDENCE_RECORD_BINDING_CONTRACT") == (
        BINDING_CONTRACT
    )

    record = R5.trade_record(monkeypatch)
    before = _policy(source=_AuthSource()).evaluate(R5.request(record))
    monkeypatch.setattr(policy_module, BINDING_GUARD, True)
    after = _policy(source=_AuthSource()).evaluate(R5.request(record))
    assert before.decision is INTEGRATION, _text(before)
    assert after.decision is ACCEPT, (
        "the guard has no runtime effect; it would be documentation wearing the "
        "name of a control: " + _text(after)
    )

def test_zz_the_imported_policy_module_is_the_expected_tree():
    """Guards every mutation result in this matrix.

    PYTHONPATH can lose to another path entry, in which case a mutated tree is
    never imported and every mutation is scored as 'survived' for the wrong
    reason.  The runner sets ``AUDIT_EXPECT_SRC_ROOT`` to the disposable copy and
    this case fails loudly if the imported module came from anywhere else.
    """
    actual = pathlib.Path(policy_module.__file__).resolve()
    expected = os.environ.get("AUDIT_EXPECT_SRC_ROOT", "").strip()
    root = (
        pathlib.Path(expected).resolve()
        if expected
        else (QUANT_LAB / "src").resolve()
    )
    assert root in actual.parents, f"imported {actual}, expected under {root}"
    expected_sha = os.environ.get("AUDIT_EXPECT_POLICY_SHA256", "").strip()
    if expected_sha:
        got = hashlib.sha256(actual.read_bytes()).hexdigest()
        assert got == expected_sha, f"policy.py hashes {got}, expected {expected_sha}"
