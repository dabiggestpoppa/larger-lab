def _decimal(value: object) -> Decimal | None:
    try:
        return Decimal(str(value))
    except (InvalidOperation, TypeError, ValueError):
        return None


def _validate(envelope: str, payload: object) -> tuple[str, object]:
    """Return (status, isolated_view).  A 200 with an error envelope is not a capture."""
    if envelope == "OKX_CODE":
        if not isinstance(payload, dict):
            return "PAYLOAD_NOT_OBJECT", None
        if str(payload.get("code")) != "0" or not payload.get("data"):
            return f"PROVIDER_ERROR_ENVELOPE(code={payload.get('code')})", None
        return "OK", payload["data"][0]
    if envelope == "DERIBIT_JSONRPC":
        if not isinstance(payload, dict):
            return "PAYLOAD_NOT_OBJECT", None
        if "error" in payload:
            return f"PROVIDER_ERROR_ENVELOPE({payload['error']})", None
        result = payload.get("result")
        if isinstance(result, list):
            result = result[0] if result else None
        return ("OK", result) if result else ("EMPTY_RESULT", None)
    if envelope == "GATE_OBJECT":
        return ("OK", payload) if isinstance(payload, dict) and "name" in payload else ("UNEXPECTED_SHAPE", None)
    if envelope == "GATE_LIST":
        if isinstance(payload, list) and payload:
            return "OK", payload[0]
        return "UNEXPECTED_SHAPE", None
    if envelope == "KRAKEN_RESULT":
        if not isinstance(payload, dict):
            return "PAYLOAD_NOT_OBJECT", None
        if payload.get("result") != "success":
            return f"PROVIDER_ERROR_ENVELOPE({payload.get('error') or payload.get('result')})", None
        return "OK", payload
    return "UNKNOWN_ENVELOPE", None


def _measured(checks: tuple[str, ...], view: object, whole: object) -> dict:
    """Arithmetic facts read out of the capture.  No provider semantics are assumed."""
    out: dict[str, object] = {}
    v = view if isinstance(view, dict) else {}

    if "oi_vs_oi_ccy" in checks or "oi_is_not_oi_ccy" in checks:
        oi, ccy = _decimal(v.get("oi")), _decimal(v.get("oiCcy"))
        out["oi"] = str(oi)
        out["oiCcy"] = str(ccy)
        out["oi_over_oiCcy"] = str(oi / ccy) if oi and ccy else None
        out["oi_equals_oiCcy"] = (oi == ccy) if oi is not None and ccy is not None else None

    if "oi_usd_vs_oi_ccy" in checks:
        usd, ccy = _decimal(v.get("oiUsd")), _decimal(v.get("oiCcy"))
        out["oiUsd_over_oiCcy"] = str(usd / ccy) if usd and ccy else None

    if "funding_time_is_after_ts" in checks:
        ft, ts = _decimal(v.get("fundingTime")), _decimal(v.get("ts"))
        out["fundingTime"] = str(ft)
        out["ts"] = str(ts)
        out["funding_time_after_ts"] = bool(ft and ts and ft > ts)

    if "next_rate_empty_not_zero" in checks:
        nxt = v.get("nextFundingRate")
        out["nextFundingRate_raw"] = nxt
        out["nextFundingRate_is_empty_string"] = nxt == ""

    if "oi_usd_implies_coin" in checks:
        oi_usd, mark = _decimal(v.get("open_interest")), _decimal(v.get("mark_price"))
        out["open_interest"] = str(oi_usd)
        out["open_interest_over_mark"] = str(oi_usd / mark) if oi_usd and mark else None

    if "gate_interval_present" in checks:
        out["funding_interval"] = v.get("funding_interval")
    if "gate_multiplier_present" in checks:
        out["quanto_multiplier"] = v.get("quanto_multiplier")
    if "gate_total_size_present" in checks:
        out["total_size"] = v.get("total_size")

    if "kraken_two_funding_fields_disagree" in checks:
        fr, rel = _decimal(v.get("fundingRate")), _decimal(v.get("relativeFundingRate"))
        out["fundingRate"] = str(fr)
        out["relativeFundingRate"] = str(rel)
        out["magnitude_ratio"] = str(fr / rel) if fr and rel else None

    if "kraken_isolation" in checks:
        tickers = whole.get("tickers") if isinstance(whole, dict) else None
        out["multiplexed_ticker_count"] = len(tickers) if isinstance(tickers, list) else None
        out["isolated_symbol"] = v.get("symbol")

    if "kraken_bid_ordering" in checks:
        bids, asks = v.get("bids") or [], v.get("asks") or []
        prices = [_decimal(level[0]) for level in bids]
        out["bid_levels"] = len(bids)
        out["ask_levels"] = len(asks)
        out["bids_ascending_worst_first"] = all(
            prices[i] < prices[i + 1] for i in range(len(prices) - 1)
        )
        out["bids_first_element"] = str(prices[0])
        out["bids_max_price"] = str(max(prices))
        out["first_element_is_best_bid"] = prices[0] == max(prices)
        out["first_element_overstates_depth"] = str(max(prices) / prices[0])
        return out

    if any(c.startswith("book") or c == "has_change_id" or c == "level_tuple_longer_than_two" for c in checks):
        bids = v.get("bids") or []
        asks = v.get("asks") or []
        out["bid_levels"] = len(bids)
        out["ask_levels"] = len(asks)
        if bids and asks:
            out["book_ordered"] = all(
                _decimal(bids[i][0]) >= _decimal(bids[i + 1][0]) for i in range(len(bids) - 1)
            ) and all(
                _decimal(asks[i][0]) <= _decimal(asks[i + 1][0]) for i in range(len(asks) - 1)
            )
            out["book_uncrossed"] = _decimal(bids[0][0]) < _decimal(asks[0][0])
            out["best_bid"] = str(bids[0][0])
            out["best_ask"] = str(asks[0][0])
        if "has_change_id" in checks:
            out["has_change_id"] = "change_id" in v
        if "level_tuple_longer_than_two" in checks:
            out["level_arity"] = sorted({len(level) for level in list(bids) + list(asks)})
    return out


def _fetch(target: dict) -> dict:
    request = urllib.request.Request(
        str(target["url"]),
        headers={"User-Agent": USER_AGENT, "Accept": "application/json"},
    )
    acquired_at = datetime.now(timezone.utc).isoformat().replace("+00:00", "Z")
    try:
        with urllib.request.urlopen(request, timeout=TIMEOUT) as response:
            status = int(response.status)
            body = response.read()
            ctype = response.headers.get("Content-Type", "")
    except urllib.error.HTTPError as exc:
        body = exc.read() if exc.fp is not None else b""
        status = int(exc.code)
        ctype = exc.headers.get("Content-Type", "") if exc.headers else ""
    except Exception as exc:
        return {
            "acquired_at": acquired_at,
            "http_status": None,
            "outcome": "TRANSPORT_FAILURE",
            "error": f"{type(exc).__name__}: {exc}",
            "raw_path": None,
            "raw_sha256": None,
            "raw_bytes": 0,
            "validation": "NOT_ATTEMPTED",
            "isolated": None,
            "measured": {},
        }

    digest = hashlib.sha256(body).hexdigest()
    RAW_DIR.mkdir(parents=True, exist_ok=True)
    path = RAW_DIR / f"{target['capture_id']}.json"
    path.write_bytes(body)

    if status != 200:
        validation, view = f"HTTP_{status}", None
    else:
        try:
            payload = json.loads(body)
        except Exception as exc:
            validation, view, payload = f"UNPARSEABLE({type(exc).__name__})", None, None
        else:
            validation, view = _validate(str(target["envelope"]), payload)
            if validation == "OK" and target.get("view_path") and isinstance(view, dict):
                for key in target["view_path"]:
                    view = view.get(key) if isinstance(view, dict) else None
                if not isinstance(view, dict):
                    validation = f"VIEW_PATH_MISSING({'.'.join(target['view_path'])})"

            if validation == "OK" and target.get("isolate"):
                isolation = target["isolate"]
                if isinstance(isolation, str):
                    # The query parameter already isolated the row; only prove it.
                    want = str((target.get("params") or {}).get(isolation, ""))
                    if not isinstance(view, dict) or str(view.get(isolation)) != want:
                        validation = f"ISOLATION_MISSING({isolation}={want})"
                else:
                    container, key, want = isolation
                    rows = payload.get(container) if isinstance(payload, dict) else None
                    view = next((r for r in (rows or []) if r.get(key) == want), None)
                    if view is None:
                        validation = f"ISOLATION_MISSING({want})"

    measured = _measured(target.get("measured_checks", ()), view, json.loads(body) if status == 200 and body[:1] in (b"{", b"[") else None)
    captured = validation == "OK"
    return {
        "acquired_at": acquired_at,
        "http_status": status,
        "content_type": ctype,
        "outcome": "CAPTURED" if captured else "PROVIDER_REFUSED_OR_UNUSABLE",
        "validation": validation,
        "error": None,
        "raw_path": str(path),
        "raw_sha256": digest,
        "raw_bytes": len(body),
        "isolated": view,
        "measured": measured,
    }
