WHAT IT DOES
------------
Directive section 2 lists nine properties the comprehensive governance
reconciliation must satisfy. Rather than hard-code what this auditor *believes*
the audits contain, every case below PARSES the live audit source with ``ast``
and evaluates the requirement against the measured allowlists. That makes the
probe pin-independent: it reports the law as it actually is at whatever commit
``AUDIT_QUANT_LAB`` points to.

Each case asserts the REQUIREMENT, not the current implementation. A case that
fails therefore reports a real gap between the requirement and the law, and it
self-clears the moment the law is amended -- no edit to this file is needed.

REQUIREMENT vs BASELINE
-----------------------
Two different kinds of red are possible here and they must not be conflated:

* ``PRE_AMENDMENT`` -- the requirement is unmet because the authorization
  amendment has not been published yet. This is the expected state before
  Agent 1's governance commit and is not a defect in anything.
* ``LAW_DEFECT`` -- the requirement is unmet by a predicate flaw that survives
  any amount of path granting (for example a suffix-variant entry being
  admitted because an exact file path is tested with ``startswith``).

Every case declares which kind it can produce, so the disposition never
mistakes a pending amendment for a broken oracle.

# ---------------------------------------------------------------------------
# AST machinery: read the law, do not assume it
# ---------------------------------------------------------------------------


def _parse(path: Path) -> ast.Module:
    assert path.is_file(), f"audit module is missing: {path}"
    return ast.parse(path.read_text(encoding="utf-8"))


def _const(node: ast.AST) -> Any:
    """Evaluate a literal allowlist expression, folding string concatenation.

    One frozen entry in the I10 tuple is built as
    ``"research/…/sensor_fabric/" + "SENSOR_FABRIC_" +
    "IMPLEMENTATION_PROGRESS.md"``.  ``ast.literal_eval`` refuses the ``BinOp``,
    so a naive extractor raises instead of reading the law -- which would show
    up as ten errors rather than as a measurement.  Only string addition is
    folded; anything genuinely dynamic still raises, because an allowlist that
    is not a literal is a finding in its own right.
    """
    if isinstance(node, ast.Constant):
        return node.value
    if isinstance(node, ast.BinOp) and isinstance(node.op, ast.Add):
        left, right = _const(node.left), _const(node.right)
        if isinstance(left, str) and isinstance(right, str):
            return left + right
        raise ValueError(f"non-string concatenation in allowlist: {left!r} + {right!r}")
    if isinstance(node, ast.Tuple):
        return tuple(_const(element) for element in node.elts)
    if isinstance(node, ast.Set):
        return {_const(element) for element in node.elts}
    if isinstance(node, ast.List):
        return [_const(element) for element in node.elts]
    raise ValueError(f"allowlist entry is not a literal: {ast.dump(node)}")


def assigned_in_function(tree: ast.Module, func_name: str, var_name: str) -> Any:
    """Literal value of ``var_name`` assigned anywhere inside ``func_name``."""
    for node in ast.walk(tree):
        if isinstance(node, (ast.FunctionDef, ast.AsyncFunctionDef)):
            if node.name != func_name:
                continue
            for sub in ast.walk(node):
                if isinstance(sub, ast.Assign):
                    for target in sub.targets:
                        if isinstance(target, ast.Name) and target.id == var_name:
                            return _const(sub.value)
    raise LookupError(f"{var_name!r} not assigned inside {func_name!r}")


def assigned_at_module_level(tree: ast.Module, var_name: str) -> Any:
    for node in tree.body:
        if isinstance(node, ast.Assign):
            for target in node.targets:
                if isinstance(target, ast.Name) and target.id == var_name:
                    return _const(node.value)
    raise LookupError(f"{var_name!r} not assigned at module level")

# ---------------------------------------------------------------------------
# The required oracle: three entry kinds, exact files, prefix dirs, no ``..``
# ---------------------------------------------------------------------------


def classify(entry: str) -> str:
    """Kind of an allowlist entry.

    The frozen tuples are not a single kind of thing, so a one-bit
    "exact or prefix" switch cannot represent them without silently revoking
    grants that already shipped.  Three kinds are measured:

    ``dir``           trailing slash -- admits everything beneath it
    ``file``          basename carries an extension -- admits only itself
    ``legacy_prefix`` bare stem with no extension -- admits by prefix
    """
    if entry.endswith("/"):
        return "dir"
    if "." in entry.rsplit("/", 1)[-1]:
        return "file"
    return "legacy_prefix"


def build_oracle(prefixes: list[str], exact: set[str] = frozenset()) -> "Oracle":
    return Oracle(list(prefixes), set(exact))


class Oracle:
    """The required admission predicate, independent of any implementation."""

    def __init__(self, prefixes: list[str], exact: set[str]) -> None:
        self.entries = [(entry, classify(entry)) for entry in prefixes]
        self.exact = set(exact)
        self.kinds = {kind for _, kind in self.entries}

    def admits(self, candidate: str) -> bool:
        # A traversal segment is refused outright: it can turn any prefix entry
        # into a grant for a tree it was never meant to cover.
        if ".." in Path(candidate).parts:
            return False
        if candidate in self.exact:
            return True
        for entry, kind in self.entries:
            if kind == "dir" and candidate.startswith(entry):
                return True
            if kind == "file" and candidate == entry:
                return True
            if kind == "legacy_prefix" and candidate.startswith(entry):
                return True
        return False

    def admits_as_prefix(self, candidate: str) -> bool:
        """What the current implementations do: every entry is a prefix."""
        if ".." in Path(candidate).parts:
            return False
        return any(candidate.startswith(entry) for entry, _ in self.entries)

def test_m5b_inventory_reconciliation(i05_prefixes: list[str], i10_allowed: list[str]) -> None:
    """Actual inventory reconciliation: what the tuples really hold.

    The D-1 packet states I10 has 25 entries.  This case records the measured

    measured = {
        "i05_entries": len(i05_prefixes),
        "i10_entries": len(i10_allowed),
        "i05_kinds": {},
        "i10_kinds": {},
        "packet_claimed_i10": 25,

    for entry in i05_prefixes:
        measured["i05_kinds"][classify(entry)] = measured["i05_kinds"].get(classify(entry), 0) + 1

        measured["i10_kinds"][classify(entry)] = measured["i10_kinds"].get(classify(entry), 0) + 1
    measured["i10_matches_packet"] = len(i10_allowed) == 25
    print("INVENTORY " + json.dumps(measured, sort_keys=True))
    # A measured inventory must at least be internally coherent: every entry
    # falls in exactly one of the three kinds.
    assert sum(measured["i05_kinds"].values()) == len(i05_prefixes)
    assert sum(measured["i10_kinds"].values()) == len(i10_allowed)
    assert set(measured["i05_kinds"]) <= {"dir", "file", "legacy_prefix"}
    assert set(measured["i10_kinds"]) <= {"dir", "file", "legacy_prefix"}

@pytest.mark.parametrize("which", ["i05", "i10"])
def test_m6_file_entries_are_admitted_by_equality_not_prefix(
    which: str, i05_prefixes: list[str], i10_allowed: list[str]
) -> None:
    """Requirement: new file authorization is exact, not prefix-based.

    LAW_DEFECT while the implementations call ``path.startswith`` on entries
    whose kind is ``file``.  This red does NOT clear by adding a path: it clears
    only when the predicate stops treating an exact-file entry as a prefix.
    """
    entries = i05_prefixes if which == "i05" else i10_allowed
    oracle = build_oracle(entries)
    offenders: list[tuple[str, str]] = []
    for entry in entries:
        if classify(entry) != "file":
            continue
        suffix_variant = entry + "_evil.py"
        if oracle.admits_as_prefix(suffix_variant):
            offenders.append((entry, suffix_variant))
    detail = ""
    if offenders:
        detail = ". Smallest counterexample: " + json.dumps(
            {"entry": offenders[0][0], "admitted": offenders[0][1]}
        )
    assert offenders == [], (
        f"LAW_DEFECT ({which}): {len(offenders)} exact-file entries are admitted "
        f"by prefix, so a suffix variant is legal{detail}"
    )


@pytest.mark.parametrize("which", ["i05", "i10"])
def test_m7_required_oracle_refuses_suffix_variants(
    which: str, i05_prefixes: list[str], i10_allowed: list[str]
) -> None:
    """Positive control for M6: the required oracle refuses what the law admits.

    If this case failed, M6's finding would be an artifact of the oracle rather
    than of the audited predicate.
    """
    entries = i05_prefixes if which == "i05" else i10_allowed
    oracle = build_oracle(entries)
    for entry in entries:
        if classify(entry) != "file":
            continue
        assert not oracle.admits(entry + "_evil.py"), entry
        assert not oracle.admits(entry + ".pyc"), entry
        assert oracle.admits(entry), f"required oracle revoked a live grant: {entry}"

def test_m10_historical_permitted_paths_remain_permitted(
    i05_prefixes: list[str], i10_allowed: list[str]
) -> None:
    """Requirement: historical permitted paths remain permitted.

    An exactness amendment that silently revokes a ``legacy_prefix`` grant would
    break checkpoints that already shipped, so every live entry must still admit
    a path beneath itself.

    for name, entries in (("i05", i05_prefixes), ("i10", i10_allowed)):
        oracle = build_oracle(entries, exact=RUNG1_EXACT_PATHS)

        assert revoked == [], f"{name}: exactness amendment revoked {revoked}"

# M11  Staged / unstaged / untracked are evaluated consistently
# ---------------------------------------------------------------------------


def test_m11_tri_state_views_agree_in_a_disposable_repository(
    i05_prefixes: list[str]
) -> None:
    """Requirement: staged, unstaged and untracked are consistently evaluated.

    A disposable git repository is built from scratch, one file is placed in
    each of the three custody views, and the SAME predicate is applied.  Staged
    admission is the one that matters: it is what a commit ships.
    """
    oracle = build_oracle(i05_prefixes)
    prefix = "src/crypto_sensor_fabric/normalization/sensors/"
    assert any(e == prefix for e in i05_prefixes), (
        "fixture precondition: the sensors/ entry must exist for this case"
    )

    with tempfile.TemporaryDirectory() as tmp:
        repo = Path(tmp)

        def git(*args: str) -> str:
            return subprocess.run(
                ["git", *args], cwd=repo, capture_output=True,
                encoding="utf-8", errors="replace", check=False,
            ).stdout

        git("init", "-q")
        git("config", "user.email", "audit@example.invalid")
        git("config", "user.name", "audit")
        (repo / "seed.txt").write_text("seed\n", encoding="utf-8")
        git("add", "seed.txt")
        git("commit", "-q", "-m", "seed")

        target_dir = repo / Path(prefix)
        target_dir.mkdir(parents=True, exist_ok=True)

        # untracked
        (target_dir / "untracked_probe.py").write_text("x = 1\n", encoding="utf-8")

        # unstaged (tracked, modified in the worktree)
        (target_dir / "unstaged_probe.py").write_text("x = 1\n", encoding="utf-8")
        git("add", str(Path(prefix) / "unstaged_probe.py"))
        git("commit", "-q", "-m", "add unstaged probe")
        (target_dir / "unstaged_probe.py").write_text("x = 2\n", encoding="utf-8")

        # staged (in the index, not yet committed)
        (target_dir / "staged_probe.py").write_text("x = 1\n", encoding="utf-8")
        git("add", str(Path(prefix) / "staged_probe.py"))

        views = {
            "staged": [
                line for line in git("diff", "--cached", "--name-only").splitlines() if line.strip()
            ],
            "unstaged": [
                line for line in git("diff", "--name-only").splitlines() if line.strip()
            ],
            "untracked": [
                line for line in
                git("ls-files", "--others", "--exclude-standard").splitlines() if line.strip()
            ],
        }

    # Every view must contain its own probe -- otherwise the fixture, not the
    # law, is what is being measured.
    assert views["staged"], "fixture failure: nothing was staged"
    assert views["unstaged"], "fixture failure: nothing was modified"
    assert views["untracked"], "fixture failure: nothing was untracked"

    verdicts = {
        view: sorted(path for path in paths if not oracle.admits(path))
        for view, paths in views.items()
    }
    unauthorized = {view: refused for view, refused in verdicts.items() if refused}
    assert unauthorized == {}, (
        "LAW_DEFECT: the custody views disagree about authorization; "
        "staged admission is what a commit ships. " + json.dumps(unauthorized, sort_keys=True)
    )

def _git_in_audit(*args: str) -> str:
    return subprocess.run(
        ["git", *args], cwd=AUDIT_QUANT_LAB, capture_output=True,
        encoding="utf-8", errors="replace", check=False,

@pytest.mark.parametrize("artifact", FROZEN_HISTORICAL_ARTIFACTS)
def test_m12_frozen_historical_artifact_is_not_rewritten(artifact: str) -> None:
    """Requirement: frozen historical artifacts remain unchanged.

    Only rewrites and deletions are refused; a new sibling artifact is a
    different question and is not conflated with this one.
    """
    rel = f"research/crypto_foundry/sensor_fabric/evidence/bloc_05/{artifact}"
    name_status = _git_in_audit("diff", "--name-status", AUTHORIZATION_BASE, "--", rel)
    lines = [line for line in name_status.splitlines() if line.strip()]
    if not lines:
        return  # absent at base and unchanged, or untouched
    rewritten = [line for line in lines if line[0] in {"M", "D", "R"}]
    assert rewritten == [], f"{artifact} was rewritten since {AUTHORIZATION_BASE[:12]}: {rewritten}"
    # A clean state is not a vacuous state: the file must actually exist.
    assert (AUDIT_QUANT_LAB / rel).is_file() or lines, artifact

# M17  Current governance verdict for the Rung 1 grant, as one measurement
# ---------------------------------------------------------------------------


def test_m17_rung1_grant_verdict_is_recorded_as_data(
    i01_allowed: set[str], i05_prefixes: list[str], i10_allowed: list[str]
) -> None:
    """The three rulings on the proposed grant, measured together.

    This is the headline of the governance challenge: what does the law as it
    stands actually say about the two Rung 1 paths?
    """
    rulings = {}
    for name, entries, exact_kind in (
        ("i01_subpackage", sorted(i01_allowed), "set"),
        ("i05_paths", i05_prefixes, "prefix"),
        ("i10_paths", i10_allowed, "prefix"),
    ):
        if exact_kind == "set":
            rulings[name] = {
                "consumer_authorized": "consumer" in i01_allowed,
                "mechanism": "set membership (already exact)",
            }
        else:
            oracle = build_oracle(entries)
            rulings[name] = {
                p: oracle.admits(p) for p in sorted(RUNG1_EXACT_PATHS)
            }
            rulings[name]["mechanism"] = "path.startswith over a mixed tuple"
            rulings[name]["entries"] = len(entries)
    print("RUNG1_GRANT_VERDICT " + json.dumps(rulings, sort_keys=True, indent=None))
    # The measurement itself must be coherent: it reports every requested path.
    for key in ("i05_paths", "i10_paths"):
        for path in RUNG1_EXACT_PATHS:
            assert path in rulings[key], (key, path)
    assert "consumer_authorized" in rulings["i01_subpackage"]
