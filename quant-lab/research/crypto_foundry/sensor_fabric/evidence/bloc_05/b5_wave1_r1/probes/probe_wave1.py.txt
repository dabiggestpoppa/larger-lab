def dump(record: object) -> str:
    return json.dumps(record, sort_keys=True, default=str, separators=(",", ":"))


def digest(record: object) -> str:
    return hashlib.sha256(dump(record).encode("utf-8")).hexdigest()


def write(name: str, payload: object) -> None:
    path = EVIDENCE / "results" / name
    path.write_text(json.dumps(payload, indent=2, sort_keys=True) + "\n", encoding="utf-8", newline="\n")
    RESULTS[name] = payload

def corpus_validation() -> None:
    manifest_path = EVIDENCE / "fixtures" / "corpus_manifest.json"
    manifest = json.loads(manifest_path.read_text(encoding="utf-8"))
    records = []
    digest_mismatches = []
    missing_metadata = []
    tier_law_violations = []

    for entry in manifest["files"]:
        raw = (EVIDENCE / entry["path"]).read_bytes()
        actual = hashlib.sha256(raw).hexdigest()
        if actual != entry["sha256"]:
            digest_mismatches.append({"fixture_id": entry["fixture_id"], "recorded": entry["sha256"], "actual": actual})
        record = json.loads(raw.decode("utf-8"))
        records.append(record)
        absent = [f for f in REQUIRED_FIELDS if not record.get(f)]
        if absent:
            missing_metadata.append({"fixture_id": record["fixture_id"], "missing": absent})
        payload_digest = hashlib.sha256(
            json.dumps(record["payload"], sort_keys=True, separators=(",", ":"), ensure_ascii=True).encode()
        ).hexdigest()
        if payload_digest != record["evidence_digest"]:
            digest_mismatches.append({"fixture_id": record["fixture_id"], "kind": "payload_digest"})
        if record["ratified_golden"] is not False or record["approval_status"] != "NOT_REVIEWED":
            tier_law_violations.append({"fixture_id": record["fixture_id"], "law": "NOT_RATIFIED"})
        if record["tier"].startswith("TIER_A") and record["acquisition_provenance"]["captured_from_provider"]:
            tier_law_violations.append({"fixture_id": record["fixture_id"], "law": "TIER_A_NOT_A_CAPTURE"})
        if record["tier"].startswith("TIER_C") and record["acquisition_provenance"]["payload_available_offline"]:
            tier_law_violations.append({"fixture_id": record["fixture_id"], "law": "TIER_C_PAYLOAD_UNAVAILABLE"})
        if record["tier"].startswith("TIER_B") and record["semantic_claim_scope"] == "PROVIDER_SEMANTICS":
            tier_law_violations.append({"fixture_id": record["fixture_id"], "law": "TIER_B_CANNOT_CLAIM_PROVIDER_SEMANTICS"})

    aggregate = hashlib.sha256(
        "".join(f"{e['path']}\0{e['sha256']}\n" for e in sorted(manifest["files"], key=lambda x: x["fixture_id"])).encode()
    ).hexdigest()

    providers = sorted({r["provider"] for r in records if r["provider"] != "MULTI"})
    assets = sorted({r["instrument"]["canonical_asset"] for r in records})
    contract_types = sorted({r["contract_type"] for r in records})
    families = sorted({r["sensor_family"] for r in records})

    coverage = {
        "provider_count": len(providers),
        "providers": providers,
        "at_least_three_providers": len(providers) >= 3,
        "btc_present": "BTC" in assets,
        "altcoin_present": any(a not in ("BTC", "ETH") or a == "ETH" for a in assets if a != "BTC"),
        "assets": assets,
        "linear_present": any("LINEAR" in c for c in contract_types),
        "inverse_present": any("INVERSE" in c for c in contract_types),
        "contract_types": contract_types,
        "sensor_families": families,
        "missing_sensor_families": [f for f in SENSOR_FAMILIES_REQUIRED if f not in families],
        "all_sensor_families_covered": all(f in families for f in SENSOR_FAMILIES_REQUIRED),
    }

    exec_ready = [
        r["fixture_id"]
        for r in records
        if r["supports_checkpoint"] in ("I10", "I11") and r["tier"].startswith("TIER_A")
    ]

    write(
        "02_corpus_validation.json",
        {
            "corpus_version": manifest["corpus_version"],
            "fixture_count": manifest["fixture_count"],
            "tier_counts": manifest["tier_counts"],
            "manifest_corpus_digest": manifest["corpus_digest"],
            "recomputed_corpus_digest": aggregate,
            "corpus_digest_matches": aggregate == manifest["corpus_digest"],
            "digest_mismatches": digest_mismatches,
            "missing_required_metadata": missing_metadata,
            "tier_law_violations": tier_law_violations,
            "coverage": coverage,
            "ratified_golden_count": sum(1 for r in records if r["ratified_golden"]),
            "fixtures_executable_against_implemented_entry_points": sorted(exec_ready),
            "verdict": (
                "CORPUS_VALID_UNRATIFIED"
                if not digest_mismatches and not missing_metadata and not tier_law_violations
                else "CORPUS_INVALID"
            ),
        },
    )

def _import_probe(module: str) -> dict:
    try:
        importlib.import_module(module)
    except ModuleNotFoundError as exc:
        return {"module": module, "importable": False, "error": str(exc)}
    return {"module": module, "importable": True, "error": None}

        "crypto_sensor_fabric.normalization.sensors.positioning",
        "crypto_sensor_fabric.normalization.sensors.basis",
        "crypto_sensor_fabric.normalization.lineage",
    ]
    entry_points = {m.rsplit(".", 1)[-1]: _import_probe(m) for m in modules}

    trade = normalize_trade(
        _trade_payload(quantity_native=Decimal("1000"), quantity_native_unit="USDT"),
        registry=_diagnostic_registry(),

    trade_authority = normalize_trade(_trade_payload(), registry=_diagnostic_registry())

    # Refusal of an unsupported USD derivation is a positive control for I12/I13.
    substrate = {

def fixture_execution() -> None:
    checks: list[dict] = []

    def check(cid: str, law: str, ok: bool, observed: object, expected: object) -> None:
        checks.append(
            {
                "check_id": cid,
                "law": law,
                "passed": bool(ok),
                "observed": observed,
                "expected": expected,
            }
        )

    # FX-A-BINANCE_USDM-BTCUSDT-TRADE-001 (diagnostic registry present)
    rec = normalize_trade(_trade_payload(), registry=_diagnostic_registry())
    check("FX-A-...-TRADE-001/1", "native price preserved", str(rec.native_values[0].native_value) == "61234.50", str(rec.native_values[0].native_value), "61234.50")
    check("FX-A-...-TRADE-001/2", "native quantity preserved", str(rec.native_values[1].native_value) == "0.250", str(rec.native_values[1].native_value), "0.250")
    check("FX-A-...-TRADE-001/3", "canonical aggressor side UNKNOWN", rec.aggressor_side.value == "UNKNOWN", rec.aggressor_side.value, "UNKNOWN")
    check(
        "FX-A-...-TRADE-001/4",
        "provisional side SELL under the documented row",
        rec.provisional_side is not None and rec.provisional_side.mapped_side.value == "SELL",
        rec.provisional_side.mapped_side.value if rec.provisional_side else None,
        "SELL",
    )
    check("FX-A-...-TRADE-001/5", "notional_usd blocked", rec.notional_usd.blocked, rec.notional_usd.blocked, True)
    check("FX-A-...-TRADE-001/6", "SEMANTICS_UNVERIFIED present", NormalizationQualityFlag.SEMANTICS_UNVERIFIED in rec.quality_flags, [f.value for f in rec.quality_flags], ["SEMANTICS_UNVERIFIED"])

    # FX-A-...-TRADE-002 mirrored boolean
    rec2 = normalize_trade(_trade_payload(native_event_id="593302", side_evidence=SideEvidence.provider_boolean("isBuyerMaker", False, provider="BINANCE_USDM")), registry=_diagnostic_registry())
    check("FX-A-...-TRADE-002/1", "mirrored provisional side BUY", rec2.provisional_side is not None and rec2.provisional_side.mapped_side.value == "BUY", rec2.provisional_side.mapped_side.value if rec2.provisional_side else None, "BUY")
    check("FX-A-...-TRADE-002/2", "mirrored canonical side UNKNOWN", rec2.aggressor_side.value == "UNKNOWN", rec2.aggressor_side.value, "UNKNOWN")

    # FX-A-...-TRADE-003 stablecoin quote
    rec3 = normalize_trade(_trade_payload(native_event_id="593303", quantity_native=Decimal("15309.00"), quantity_native_unit="USDT"), registry=_diagnostic_registry())
    check("FX-A-...-TRADE-003/1", "stablecoin USD conversion unavailable flag", NormalizationQualityFlag.STABLECOIN_CONVERSION_UNAVAILABLE in rec3.quality_flags, [f.value for f in rec3.quality_flags], ["STABLECOIN_CONVERSION_UNAVAILABLE"])
    check("FX-A-...-TRADE-003/2", "USD twin not emitted", rec3.notional_usd.value is None, rec3.notional_usd.value, None)
    check("FX-A-...-TRADE-003/3", "unresolved citation names the units methodology", rec3.notional_usd.unresolved_citation == I08_UNITS_METHODOLOGY_ID, rec3.notional_usd.unresolved_citation, I08_UNITS_METHODOLOGY_ID)

    # FX-B-TRADE-AGGRESSOR-FROM-PRICE-DIRECTION-001
    rec4 = normalize_trade(_trade_payload(native_event_id="593304", side_evidence=SideEvidence.provider_boolean("isBuyerMaker", True, provider="BINANCE_USDM")))
    check("FX-B-TRADE-AGGRESSOR/1", "price direction cannot admit a side", rec4.aggressor_side.value == "UNKNOWN", rec4.aggressor_side.value, "UNKNOWN")

    # FX-B-LIQ-POSITION-EXECUTION-COLLAPSE-001
    liq = normalize_liquidation(_liquidation_payload(position_side_liquidated=LiquidationPositionSide.UNKNOWN, execution_side=AggressorSide.UNKNOWN))
    check("FX-B-LIQ-COLLAPSE/1", "collapsed side leaves position UNKNOWN", liq.position_side_liquidated.value == "UNKNOWN", liq.position_side_liquidated.value, "UNKNOWN")
    check("FX-B-LIQ-COLLAPSE/2", "collapsed side leaves execution UNKNOWN", liq.execution_side.value == "UNKNOWN", liq.execution_side.value, "UNKNOWN")

    # FX-A-GATE liquidation shape: position LONG with SELL execution, no authority
    liq2 = normalize_liquidation(_liquidation_payload())
    check("FX-A-LIQ-SEPARATION/1", "position side preserved provisionally", liq2.provisional_position_side.value == "LONG", liq2.provisional_position_side.value, "LONG")
    check("FX-A-LIQ-SEPARATION/2", "execution side preserved provisionally", liq2.provisional_execution_side.value == "SELL", liq2.provisional_execution_side.value, "SELL")
    check("FX-A-LIQ-SEPARATION/3", "canonical position side not promoted", liq2.position_side_liquidated.value == "UNKNOWN", liq2.position_side_liquidated.value, "UNKNOWN")
    check("FX-A-LIQ-SEPARATION/4", "canonical execution side not promoted", liq2.execution_side.value == "UNKNOWN", liq2.execution_side.value, "UNKNOWN")
    check("FX-A-LIQ-SEPARATION/5", "zero interval state not applicable", liq2.zero_interval_state is LiquidationZeroIntervalState.NOT_APPLICABLE, liq2.zero_interval_state.value, "NOT_APPLICABLE")
    check("FX-A-LIQ-SEPARATION/6", "liquidation_usd blocked", liq2.liquidation_usd.blocked, liq2.liquidation_usd.blocked, True)

    passed = sum(1 for c in checks if c["passed"])
    write(

# 05 - I21 property controls
# ---------------------------------------------------------------------------


def i21_properties() -> None:
    controls: list[dict] = []

    def control(cid: str, law: str, ok: bool, detail: object) -> None:
        controls.append({"control_id": cid, "law": law, "passed": bool(ok), "detail": detail})

    payload = _trade_payload()
    first = normalize_trade(payload, registry=_diagnostic_registry())
    second = normalize_trade(payload, registry=_diagnostic_registry())
    control("P1", "identical input yields byte-identical record", dump(first) == dump(second), {"digest": digest(first)})

    interleaved = normalize_trade(_trade_payload(native_event_id="999999"), registry=_diagnostic_registry())
    third = normalize_trade(payload, registry=_diagnostic_registry())
    control(
        "P2",
        "execution-order change does not alter a record",
        dump(third) == dump(first),
        {"interleaved_digest": digest(interleaved), "repeat_digest": digest(third)},
    )

    rec = first
    control(
        "P3",
        "native value survival: price and quantity unchanged",
        str(rec.native_values[0].native_value) == str(payload.price_native)
        and str(rec.native_values[1].native_value) == str(payload.quantity_native),
        {"price": str(rec.native_values[0].native_value), "quantity": str(rec.native_values[1].native_value)},
    )
    control(
        "P4",
        "null is not zero: blocked amounts are None",
        rec.notional_usd.value is None and rec.notional_usd.blocked is True,
        {"value": rec.notional_usd.value, "blocked": rec.notional_usd.blocked, "reason": rec.notional_usd.missingness_reason.value},
    )
    declared_order = list(NormalizationQualityFlag)
    control(
        "P5",
        "canonical flags are unique and follow the declared canonical order",
        len(set(rec.quality_flags)) == len(rec.quality_flags)
        and list(rec.quality_flags)
        == sorted(rec.quality_flags, key=declared_order.index),
        {
            "flags": [f.value for f in rec.quality_flags],
            "declared_indices": [declared_order.index(f) for f in rec.quality_flags],
        },
    )

    other_venue = normalize_trade(_trade_payload(provider="GATE_FUTURES", venue="GATE", native_instrument="BTC_USDT"), registry=_diagnostic_registry())
    control(
        "P6",
        "no cross-venue dedupe: same economics, different venue stays distinct",
        dump(other_venue) != dump(rec),
        {"binance": digest(rec), "gate": digest(other_venue)},
    )

    no_reg = normalize_trade(_trade_payload())
    diag = normalize_trade(_trade_payload(), registry=_diagnostic_registry())
    control(
        "P7",
        "registry shape is not authority: canonical side UNKNOWN in both cases",
        no_reg.aggressor_side.value == "UNKNOWN" and diag.aggressor_side.value == "UNKNOWN",
        {"without_registry": no_reg.aggressor_side.value, "diagnostic_registry": diag.aggressor_side.value},
    )

    liq = normalize_liquidation(_liquidation_payload())
    control(
        "P8",
        "liquidation null is not zero: blocked USD stays None",
        liq.liquidation_usd.value is None and liq.liquidation_usd.blocked,
        {"value": liq.liquidation_usd.value, "blocked": liq.liquidation_usd.blocked},
    )

    # Offline purity: no network, no wall clock in the executable normalizers.
    purity_targets = {
        "trade": SRC / "crypto_sensor_fabric/normalization/sensors/trade.py",
        "liquidation": SRC / "crypto_sensor_fabric/normalization/sensors/liquidation.py",
        "terms_conversion": SRC / "crypto_sensor_fabric/normalization/terms/conversion.py",
    }
    banned = [r"\bimport\s+requests\b", r"\bimport\s+socket\b", r"\bimport\s+urllib\b", r"\btime\.time\(", r"\bdatetime\.now\(", r"\butcnow\("]
    purity = {}
    for label, path in purity_targets.items():
        text = path.read_text(encoding="utf-8")
        code_lines = [ln for ln in text.splitlines() if not ln.strip().startswith("#")]
        body = "\n".join(code_lines)
        hits = [b for b in banned if re.search(b, body)]
        purity[label] = {"violations": hits, "clean": not hits}
    control("P9", "normalizer code is network-free and clock-free", all(v["clean"] for v in purity.values()), purity)

    control(
        "P10",
        "reversal: a record cannot hold a canonical amount with value None",
        rec.notional_usd.blocked == (rec.notional_usd.value is None),
        {"blocked": rec.notional_usd.blocked, "value": rec.notional_usd.value},
    )

    passed = sum(1 for c in controls if c["passed"])
    write(
        "05_i21_property_controls.json",
        {
            "control_count": len(controls),
            "passed": passed,
            "failed": len(controls) - passed,
            "controls": controls,
            "scope": "properties executable against I10/I11 today; I12-I15 properties are declared, not claimed",
            "determinism_digest": digest({"controls": controls}),
            "verdict": "PROPERTIES_HOLD" if passed == len(controls) else "PROPERTY_FAILURE",
        },
    )

# 06 - I16 lineage / deterministic-ID preparation
# ---------------------------------------------------------------------------


def i16_preparation() -> None:
    rec = normalize_trade(_trade_payload(), registry=_diagnostic_registry())
    fields = sorted(type(rec).model_fields.keys())
    id_like = [f for f in fields if f in ("t1_record_id", "record_id", "id")]
    lineage_fields = sorted(type(rec.lineage).model_fields.keys())

    # The property a deterministic ID must have, measured on record content.
    d1 = digest(rec)
    d2 = digest(normalize_trade(_trade_payload(), registry=_diagnostic_registry()))
    generation_is_passthrough = rec.normalization_generation == "gen-wave1-1"

    # Conflicting source references: two distinct blobs cannot collapse to one lineage.
    conflicting = normalize_trade(_trade_payload(evidence_blob_hashes=(BLOB,)), registry=_diagnostic_registry())
    try:
        normalize_trade(
            _trade_payload(evidence_blob_hashes=(BLOB, BLOB)), registry=_diagnostic_registry()
        )
        duplicate_ref_refused = False
    except ValueError:
        duplicate_ref_refused = True

    # Future revisions: a revision id change must change content, not silently reuse it.
    revised = normalize_trade(_trade_payload(source_revision_id="rev-wave1-2"), registry=_diagnostic_registry())

    write(
        "06_i16_lineage_id_preparation.json",
        {
            "state": "BLOCKED_PENDING_IMPLEMENTATION",
            "lineage_package_present": (SRC / "crypto_sensor_fabric/normalization/lineage").exists(),
            "record_has_deterministic_id_field": bool(id_like),
            "id_like_fields_found": id_like,
            "lineage_fields_present": lineage_fields,
            "lineage_completeness": {
                "raw_projection_refs": bool(rec.lineage.raw_projection_refs),
                "acquisition_refs": bool(rec.lineage.acquisition_refs),
                "evidence_blob_hashes": bool(rec.lineage.evidence_blob_hashes),
                "code_version": rec.lineage.code_version,
                "config_version": rec.lineage.config_version,
                "conversion_lineage_refs": list(rec.lineage.conversion_lineage_refs),
                "lineage_state": rec.lineage_state.value,
            },
            "content_digest_stability": {
                "first": d1,
                "second": d2,
                "stable": d1 == d2,
                "note": "computed externally; the production tree mints no T1 id yet",
            },
            "generation_is_caller_passthrough": generation_is_passthrough,
            "identical_blob_refs_refused": duplicate_ref_refused,
            "conflicting_source_refs_retained_separately": d1 != digest(conflicting),
            "revision_change_alters_content": d1 != digest(revised),
            "lineage_gate_claimable": False,
            "lineage_gate_reason": (
                "no lineage package, no deterministic T1 id, and only 2 of 8 sensor families "
                "implemented; LINEAGE_GATE must not be claimed on a partial sensor-family set"
            ),
            "prepared_controls": [
                "repeated identical inputs -> identical content digest",
                "execution-order change -> unchanged content digest",
                "machine-local state change -> unchanged content digest (cross-process repeat run)",
                "different providers with similar observations -> distinct records",
                "missing T0 evidence -> input model refuses (min_length=1 refs)",
                "conflicting source references -> retained separately, never merged",
                "future revision -> new content digest, prior record immutable by construction",
                "generation immutability -> enforced only once I18 exists; declared, not claimed",
            ],
        },
    )
