"""SENSOR-B5-WAVE1-R1 - I20 golden-fixture corpus authoring (offline, deterministic).

Writes one JSON file per fixture plus ``corpus_manifest.json`` into
``<evidence>/fixtures/provider`` and ``<evidence>/fixtures/synthetic``.

Three tiers are kept structurally separate and are NEVER mixed:

    TIER_A_DOCUMENTED_FIELD_SHAPE     provider field vocabulary reproduced from the
                                      frozen in-repo documentation. Payload VALUES
                                      are authored, so the tier proves field-shape
                                      handling only -- never provider semantics.
    TIER_B_SYNTHETIC_ADVERSARIAL      deliberately hostile constructions. They can
                                      prove refusal/reconstruction BEHAVIOUR, never
                                      actual provider semantics.
    TIER_C_PROVIDER_EVIDENCE_REFERENCED  records that point at provider evidence
                                      actually recorded in this repository. The
                                      payload bytes are NOT available offline, so
                                      these records are unusable as inputs and are
                                      marked as such.

Every record carries the twelve metadata fields the mission requires. No record is
ratified golden evidence: ``approval_status`` is ``NOT_REVIEWED`` and
``ratified_golden`` is false everywhere. Nothing here is a provider approval.

No network, no wall clock, no uuid, no randomness: byte-identical output for
byte-identical source.
"""

RECORDS: list[dict] = []


def _digest(payload: object) -> str:
    blob = json.dumps(payload, sort_keys=True, separators=(",", ":"), ensure_ascii=True)
    return hashlib.sha256(blob.encode("utf-8")).hexdigest()


def fx(
    *,
    fixture_id: str,
    tier: str,
    provider: str,
    instrument: dict,
    sensor_family: str,
    source: dict,
    native_units: dict,
    timestamp_semantics: dict,
    semantic_claim: str,
    semantic_claim_scope: str,
    payload: dict,
    expectation: dict,
    known_limitations: list[str],
    supports_checkpoint: str,
    acquisition: dict | None = None,
    synthetic: bool = True,
    checkpoint_executable: bool = False,
) -> None:
    RECORDS.append(
        {
            "fixture_id": fixture_id,
            "corpus_version": CORPUS_VERSION,
            "tier": tier,
            "synthetic": synthetic,
            "provider": provider,
            "venue": provider,
            "instrument": instrument,
            "sensor_family": sensor_family,
            "contract_type": instrument["contract_type"],
            "source": source,
            "acquisition_provenance": acquisition
            or {
                "method": "DOCUMENTED_SHAPE_RECONSTRUCTION",
                "network_calls": 0,
                "captured_from_provider": False,
                "authored_by": "second-agent (B5-WAVE1-R1)",
                "payload_values": "AUTHORED_NOT_CAPTURED",
            },
            "native_units": native_units,
            "timestamp_semantics": timestamp_semantics,
            "semantic_claim": semantic_claim,
            "semantic_claim_scope": semantic_claim_scope,
            "payload": payload,
            "expectation": expectation,
            "evidence_digest": _digest(payload),
            "known_limitations": known_limitations,
            "approval_status": "NOT_REVIEWED",
            "ratified_golden": False,
            "supports_checkpoint": supports_checkpoint,
            "checkpoint_executable_today": (
                checkpoint_executable or supports_checkpoint in ("I10", "I11")
            ),
        }
    )


def docs(provider: str, endpoint: str, field_confidence: str = "SECONDARY_DOCUMENTED") -> dict:
    return {
        "kind": "DOCUMENTED_FIELD_SHAPE",
        "endpoint": endpoint,
        "documentation_ref": DOCS,
        "provider_evidence_ref": PROVIDER_DOC[provider],
        "evidence_commit": EVIDENCE_COMMIT,
        "field_name_confidence": field_confidence,
    }


def ts(field: str, meaning: str, precision: str = "MILLISECOND", interval: bool = False) -> dict:
    return {
        "timestamp_field": field,
        "meaning": meaning,
        "timezone": "UTC",
        "source_precision": precision,
        "interval_close_based": interval,
        "market_available_at_established": False,
        "pit_status": "NOT_ESTABLISHED",
    }

def main() -> None:
    for tier_dir, tier in (
        ("provider", ("TIER_A_DOCUMENTED_FIELD_SHAPE", "TIER_C_PROVIDER_EVIDENCE_REFERENCED")),
        ("synthetic", ("TIER_B_SYNTHETIC_ADVERSARIAL",)),
    ):
        out = EVIDENCE / "fixtures" / tier_dir
        out.mkdir(parents=True, exist_ok=True)
        for record in RECORDS:
            if record["tier"] not in tier:
                continue
            path = out / f"{record['fixture_id']}.json"
            path.write_text(
                json.dumps(record, indent=2, sort_keys=True) + "\n",
                encoding="utf-8",
                newline="\n",
            )

    ids = [r["fixture_id"] for r in RECORDS]
    assert len(ids) == len(set(ids)), "duplicate fixture ids"

    files = []
    for record in RECORDS:
        sub = "provider" if record["tier"].startswith(("TIER_A", "TIER_C")) else "synthetic"
        rel = f"fixtures/{sub}/{record['fixture_id']}.json"
        data = (EVIDENCE / rel).read_bytes()
        files.append(
            {
                "fixture_id": record["fixture_id"],
                "path": rel,
                "sha256": hashlib.sha256(data).hexdigest(),
                "tier": record["tier"],
                "provider": record["provider"],
                "sensor_family": record["sensor_family"],
                "contract_type": record["contract_type"],
                "canonical_asset": record["instrument"]["canonical_asset"],
                "supports_checkpoint": record["supports_checkpoint"],
                "ratified_golden": False,
                "approval_status": record["approval_status"],
            }
        )
    files.sort(key=lambda f: f["fixture_id"])
    aggregate = hashlib.sha256(
        "".join(f"{f['path']}\0{f['sha256']}\n" for f in files).encode("utf-8")
    ).hexdigest()

    providers = sorted({r["provider"] for r in RECORDS})
    manifest = {
        "corpus_version": CORPUS_VERSION,
        "authority_commit": AUTHORITY_COMMIT,
        "evidence_commit": EVIDENCE_COMMIT,
        "fixture_count": len(RECORDS),
        "tier_counts": {
            tier: sum(1 for r in RECORDS if r["tier"] == tier)
            for tier in (
                "TIER_A_DOCUMENTED_FIELD_SHAPE",
                "TIER_B_SYNTHETIC_ADVERSARIAL",
                "TIER_C_PROVIDER_EVIDENCE_REFERENCED",
            )
        },
        "providers": providers,
        "provider_count": len([p for p in providers if p != "MULTI"]),
        "assets": sorted({r["instrument"]["canonical_asset"] for r in RECORDS}),
        "contract_types": sorted({r["contract_type"] for r in RECORDS}),
        "sensor_families": sorted({r["sensor_family"] for r in RECORDS}),
        "ratified_golden_count": 0,
        "approval_status": "NOT_REVIEWED",
        "corpus_digest": aggregate,
        "digest_algorithm": "sha256 over sorted '<path>\\0<sha256>\\n' rows",
        "files": files,
    }
    (EVIDENCE / "fixtures" / "corpus_manifest.json").write_text(
        json.dumps(manifest, indent=2, sort_keys=True) + "\n",
        encoding="utf-8",
        newline="\n",
    )
    print(json.dumps({k: manifest[k] for k in ("fixture_count", "tier_counts", "provider_count", "corpus_digest")}, indent=2))


if __name__ == "__main__":
    main()
