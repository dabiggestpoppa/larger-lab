import pandas as pd, numpy as np
def run_eth_occ_test(csv_path, buffers=[5,7,10]):
    df = pd.read_csv(csv_path, header=None,
        names=['dt','open','high','low',
               'close','vol','spread','ts'])
    df['dt'] = pd.to_datetime(df['dt'],
        format='%Y.%m.%d %H:%M:%S')
    df = df.set_index('dt').sort_index()
    df = (df.tz_localize('UTC') if df.index.tz
          is None else df).tz_convert(
          'America/New_York')
    df['session'] = (df.index -
        pd.Timedelta(hours=3)).dt.floor('D')
    res = {b:{'wr':[],'rm':[],'ws':0,'cs':0}
           for b in buffers}
    for day, data in df.groupby('session'):
        asian = data[(data.index.hour>=19)|
                     (data.index.hour<3)]
        if len(asian)<10: continue
        ar = asian['high'].max()-asian['low'].min()
        if ar<70:    t,au="T1",35
        elif ar<105: t,au="T2",42
        elif ar<160: t,au="T3",52
        else: continue
        sess = data[(data.index.hour>=3)&
                    (data.index.hour<12)]
        if sess.empty: continue
        state,orig,dir_='SEARCH',None,0
        for i,r in sess.iterrows():
          if state=='SEARCH':
            if orig is None: orig=r['close']
            up=r['high']-orig
            dn=orig-r['low']
            if up>=au and r['close']<r['open']:
              state,dir_,orig='PB',-1,r['high']
            elif dn>=au and r['close']>r['open']:
              state,dir_,orig='PB',1,r['low']
elif state=='PB':
            dz_l=(orig-au*1.2 if dir_==-1
                  else orig+au*0.8)
            dz_u=(orig-au*0.8 if dir_==-1
                  else orig+au*1.2)
            opp=((r['close']>r['open'])
                 if dir_==-1
                 else (r['close']<r['open']))
            if dz_l<=r['close']<=dz_u and opp:
              entry=r['close']
              occ=(r['high'] if dir_==-1
                   else r['low'])
              tgt=entry+(au*dir_)
              fut=sess[sess.index>i]
              if fut.empty:
                state,orig='SEARCH',None
                continue
              for buf in buffers:
                stop=occ+(buf*dir_)
                tp,cs,ws=False,False,False
                for _,f in fut.iterrows():
                  if ((dir_==1 and
                       f['close']>=tgt) or
                      (dir_==-1 and
                       f['close']<=tgt)):
                    tp=True; break
                  if ((dir_==1 and
                       f['close']<=stop) or
                      (dir_==-1 and
                       f['close']>=stop)):
                    cs=True; break
                  if ((dir_==1 and
                       f['low']<=stop) or
                      (dir_==-1 and
                       f['high']>=stop)):
                    ws=True
                if tp:
                  res[buf]['wr'].append(1)
                  rsk=abs(entry-stop)
                  rwd=abs(tgt-entry)
                  res[buf]['rm'].append(
                    rwd/rsk if rsk>0 else 0)
                elif cs:
                  res[buf]['wr'].append(0)
                  res[buf]['cs']+=1
                elif ws: res[buf]['ws']+=1
              state,orig='SEARCH',None
    return {b:{'hit_rate':
      round(np.mean(d['wr'])*100,1),
      'avg_r':round(np.mean(d['rm']),2)}
      for b,d in res.items()}

from sklearn.cluster import KMeans
import numpy as np, pandas as pd
def discover_tiers(csv_path, start='2024-01-01', end='2026-05-08', pip_size=0.0001):
    df = pd.read_csv(csv_path, header=None,
        names=['dt','open','high','low','close','vol','spread','ts'])
    df['dt'] = pd.to_datetime(df['dt'])
    df = df.set_index('dt').sort_index()
    df = df.tz_localize('UTC').tz_convert('America/New_York')
    df['session'] = (df.index - pd.Timedelta(hours=3)).dt.floor('D')
    ranges = []
    for day, data in df.groupby('session'):
        asian = data[(data.index.hour>=19)|(data.index.hour<3)]
        if len(asian) >= 10:
            ar = (asian['high'].max()-asian['low'].min())/pip_size
            ranges.append(ar)
    ranges = np.array(ranges).reshape(-1,1)
    km = KMeans(n_clusters=3, random_state=42, n_init=10).fit(ranges)
    c = sorted(km.cluster_centers_.flatten())
    return {
        'cutoffs': ((c[0]+c[1])/2, (c[1]+c[2])/2),
        'T1': {'au': int(c[0]*0.5), 'trig': int(c[0]*0.5*1.2)},
        'T2': {'au': int(c[1]*0.5), 'trig': int(c[1]*0.5*1.2)},
