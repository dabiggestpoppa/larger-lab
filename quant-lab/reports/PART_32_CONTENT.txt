import pandas as pd, numpy as np
def run_eth_occ_test(csv_path, buffers=[5,7,10]):
    df = pd.read_csv(csv_path, header=None,
        names=['dt','open','high','low',
               'close','vol','spread','ts'])
    df['dt'] = pd.to_datetime(df['dt'],
        format='%Y.%m.%d %H:%M:%S')
    df = df.set_index('dt').sort_index()
    df = (df.tz_localize('UTC') if df.index.tz
          is None else df).tz_convert(
          'America/New_York')
    df['session'] = (df.index -
        pd.Timedelta(hours=3)).dt.floor('D')
    res = {b:{'wr':[],'rm':[],'ws':0,'cs':0}
           for b in buffers}
    for day, data in df.groupby('session'):
        asian = data[(data.index.hour>=19)|
                     (data.index.hour<3)]
        if len(asian)<10: continue
        ar = asian['high'].max()-asian['low'].min()
        if ar<70:    t,au="T1",35

        else: continue

          if state=='SEARCH':
            if orig is None: orig=r['close']

elif state=='PB':

UNIVERSAL TIER DISCOVERY — K-MEANS METHODOLOGY
Every instrument below uses the same tier discovery process. Asian Range data from 7PM-3AM EST is clustered via k-means into 3 natural
volatility groups. Atomic Unit = ~50% of tier centroid. Tier Trigger = AU x 1.2. Density Zone = AU ±20%. These are structural parameters
derived from data — not assumed from forex scaling.
# UNIVERSAL TIER DISCOVERY TEMPLATE (run on any M5 CSV)
from sklearn.cluster import KMeans
import numpy as np, pandas as pd
def discover_tiers(csv_path, start='2024-01-01', end='2026-05-08', pip_size=0.0001):
    df = pd.read_csv(csv_path, header=None,
        names=['dt','open','high','low','close','vol','spread','ts'])
    df['dt'] = pd.to_datetime(df['dt'])
    df = df.set_index('dt').sort_index()
    df = df.tz_localize('UTC').tz_convert('America/New_York')
    df['session'] = (df.index - pd.Timedelta(hours=3)).dt.floor('D')
    ranges = []
    for day, data in df.groupby('session'):
        asian = data[(data.index.hour>=19)|(data.index.hour<3)]
        if len(asian) >= 10:
            ar = (asian['high'].max()-asian['low'].min())/pip_size
            ranges.append(ar)
    ranges = np.array(ranges).reshape(-1,1)
    km = KMeans(n_clusters=3, random_state=42, n_init=10).fit(ranges)
    c = sorted(km.cluster_centers_.flatten())
    return {
        'cutoffs': ((c[0]+c[1])/2, (c[1]+c[2])/2),
        'T1': {'au': int(c[0]*0.5), 'trig': int(c[0]*0.5*1.2)},
        'T2': {'au': int(c[1]*0.5), 'trig': int(c[1]*0.5*1.2)},
        'T3': {'au': int(c[2]*0.5), 'trig': int(c[2]*0.5*1.2)},
    }
