CODE 1 ΓÇö ATOMIC UNIT DISCOVERY (K-MEANS CLUSTERING)
import pandas as pd
import numpy as np
from sklearn.cluster import KMeans
def discover_atomic_units(csv_path, pip_size=0.0001):
    df = pd.read_csv(csv_path, header=None,
                     names=['dt','open','high','low','close','vol','spread','ts'])
    df['dt'] = pd.to_datetime(df['dt'])
    df = df.set_index('dt').sort_index()
    if df.index.tz is None:
        df = df.tz_localize('UTC').tz_convert('America/New_York')
    else:
        df = df.tz_convert('America/New_York')
    df['session'] = (df.index - pd.Timedelta(hours=3)).dt.floor('D')
    ranges = []
    for day, data in df.groupby('session'):
        asian = data[(data.index.hour >= 19) | (data.index.hour < 3)]
        if len(asian) >= 10:
            ar = (asian['high'].max() - asian['low'].min()) / pip_size
            ranges.append(ar)
    ranges = np.array(ranges).reshape(-1, 1)
    km = KMeans(n_clusters=3, random_state=42, n_init=10).fit(ranges)
    centroids = sorted(km.cluster_centers_.flatten())
    cutoff1 = (centroids[0] + centroids[1]) / 2
    cutoff2 = (centroids[1] + centroids[2]) / 2
    results = {
        'T1': {'range_max': cutoff1, 'centroid': centroids[0],
               'atomic_unit': int(centroids[0] * 0.50),
               'trigger': int(centroids[0] * 0.50 * 1.2)},
        'T2': {'range_min': cutoff1, 'range_max': cutoff2, 'centroid': centroids[1],
               'atomic_unit': int(centroids[1] * 0.50),
               'trigger': int(centroids[1] * 0.50 * 1.2)},
        'T3': {'range_min': cutoff2, 'centroid': centroids[2],
               'atomic_unit': int(centroids[2] * 0.50),
               'trigger': int(centroids[2] * 0.50 * 1.2)}
    }
    return results
# USAGE: discover_atomic_units('EURUSD_M5.csv', pip_size=0.0001)

def validate_atomic_loop(csv_path, pip_size, tier_configs):
    df = pd.read_csv(csv_path, header=None, names=['dt','open','high','low','close','vol','spread','ts'])
    df['dt'] = pd.to_datetime(df['dt'])
    df = df.set_index('dt').sort_index().tz_localize('UTC').tz_convert('America/New_York')
    df['session'] = (df.index - pd.Timedelta(hours=3)).dt.floor('D')
    hits, misses = 0, 0
    for day, data in df.groupby('session'):
        asian = data[(data.index.hour >= 19) | (data.index.hour < 3)]
        if len(asian) < 10: continue
        ar = (asian['high'].max() - asian['low'].min()) / pip_size
        if ar < tier_configs['T1']['range_max']: tier, au = 'T1', tier_configs['T1']['atomic_unit']
        elif ar < tier_configs['T2']['range_max']: tier, au = 'T2', tier_configs['T2']['atomic_unit']
        else: tier, au = 'T3', tier_configs['T3']['atomic_unit']
        trigger = au * 1.2
        sess = data[(data.index.hour >= 3) & (data.index.hour < 12)]
        if sess.empty: continue
        state, origin = 'SEARCH', None
        for i, r in sess.iterrows():
            if state == 'SEARCH':
